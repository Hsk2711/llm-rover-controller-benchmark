    private (float speed01, float turn) ComputeSeekGoalCommand(
        float goalAngle,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        // Repulsive push from nearby obstacles on left/right so rover stays centered/safe
        float avoidanceBias = 0.0f;
        if (left < 0.55f)
            avoidanceBias -= (0.55f - left) * 1.6f;
        if (right < 0.55f)
            avoidanceBias += (0.55f - right) * 1.6f;
        if (leftSide < 0.28f)
            avoidanceBias -= (0.28f - leftSide) * 1.2f;
        if (rightSide < 0.28f)
            avoidanceBias += (0.28f - rightSide) * 1.2f;

        float goalTurn = goalAngle / 42.0f;
        float turn = Mathf.Clamp(goalTurn + avoidanceBias, -1.0f, 1.0f);

        float angleFactor = Mathf.Clamp01(1.0f - Mathf.Abs(goalAngle) / 135.0f);
        float clearanceFactor = Mathf.Clamp01((front - 0.22f) / 1.10f);
        float speed01 = Mathf.Lerp(0.25f, 1.0f, angleFactor * clearanceFactor);

        if (Mathf.Abs(goalAngle) > 75.0f)
            speed01 = Mathf.Min(speed01, 0.35f);

        return (speed01, turn);
    }

    private (float speed01, float turn) ComputeWallFollowCommand(
        bool followLeftWall,
        float front,
        float diagAhead,
        float diagOpposite,
        float sideWall,
        float oppositeSideWall)
    {
        // Sign convention: +1 = turn left, -1 = turn right
        float towardWallSign = followLeftWall ? +1.0f : -1.0f;
        float awayWallSign = -towardWallSign;

        const float desiredSideDist = 0.34f;
        float turn = 0.0f;
        float speed01 = 0.70f;

        // 1. Obstacle ahead or on the diagonal toward the followed wall -> turn away from wall
        if (front < 0.55f || diagAhead < 0.45f)
        {
            float urgency = Mathf.Clamp01(1.0f - Mathf.Min(front / 0.55f, diagAhead / 0.45f));
            turn = awayWallSign * Mathf.Lerp(0.45f, 1.0f, urgency);
            speed01 = Mathf.Lerp(0.55f, 0.20f, urgency);
        }
        // 2. Side sensor detects wall -> hold target standoff distance
        else if (sideWall < 1.35f)
        {
            float error = sideWall - desiredSideDist; // positive = too far from wall, negative = too close
            turn = Mathf.Clamp(towardWallSign * (error * 2.4f), -0.65f, 0.65f);
            speed01 = 0.75f;
        }
        // 3. Lost side wall (e.g., rounded an outer corner) -> curve gently back toward wall to find it
        else
        {
            turn = towardWallSign * 0.42f;
            speed01 = 0.65f;
        }

        // Corridor / opposite-side safety adjustment
        if (oppositeSideWall < 0.24f)
        {
            turn += towardWallSign * ((0.24f - oppositeSideWall) * 1.5f);
        }
        if (diagOpposite < 0.28f)
        {
            turn += towardWallSign * ((0.28f - diagOpposite) * 1.4f);
        }

        return (speed01, Mathf.Clamp(turn, -1.0f, 1.0f));
    }

    private (float speed01, float turn) ComputeEscapeCommand(float front, float left, float right)
    {
        // Pivot in place or barely creep forward while turning out of a tight spot / dead end
        float turn = _escapeSign * 1.0f;
        float speed01 = (front > 0.45f && left > 0.35f && right > 0.35f) ? 0.22f : 0.0f;
        return (speed01, turn);
    }

    private float ApplyProximitySpeedLimit(
        float speed01,
        float turn,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        float minAll = Mathf.Min(front, Mathf.Min(left, right));
        float minSide = Mathf.Min(leftSide, rightSide);

        if (minAll < 0.20f || minSide < 0.14f)
            return Mathf.Min(speed01, 0.12f);

        if (minAll < 0.34f || minSide < 0.20f)
            speed01 = Mathf.Min(speed01, 0.35f);
        else if (minAll < 0.55f)
            speed01 = Mathf.Min(speed01, 0.65f);

        // Reduce forward speed during sharp turns so turning radius stays tight
        float absTurn = Mathf.Abs(turn);
        if (absTurn > 0.75f)
            speed01 = Mathf.Min(speed01, 0.30f);
        else if (absTurn > 0.45f)
            speed01 = Mathf.Min(speed01, 0.55f);

        return speed01;
    }

    private bool IsGoalPathClear(
        float goalAngle,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        if (Mathf.Abs(goalAngle) <= 22.0f)
            return front > 0.65f && left > 0.45f && right > 0.45f;

        if (goalAngle > 22.0f && goalAngle <= 68.0f)
            return left > 0.65f && front > 0.45f;

        if (goalAngle < -22.0f && goalAngle >= -68.0f)
            return right > 0.65f && front > 0.45f;

        if (goalAngle > 68.0f && goalAngle <= 115.0f)
            return leftSide > 0.65f && left > 0.45f;

        if (goalAngle < -68.0f && goalAngle >= -115.0f)
            return rightSide > 0.65f && right > 0.45f;

        return false;
    }

    private void UpdateProgressTracking(float goalDist)
    {
        if (goalDist + 0.04f < _bestGoalDist)
        {
            _bestGoalDist = goalDist;
            _ticksWithoutProgress = 0;
        }
        else if (Mathf.Abs(goalDist - _prevGoalDist) < 0.004f)
        {
            _ticksWithoutProgress++;
        }
        else
        {
            _ticksWithoutProgress++;
        }

        _prevGoalDist = goalDist;
    }

    private void PushSample(float f, float l, float r, float ls, float rs)
    {
        _frontBuf[_bufIndex] = f;
        _leftBuf[_bufIndex] = l;
        _rightBuf[_bufIndex] = r;
        _leftSideBuf[_bufIndex] = ls;
        _rightSideBuf[_bufIndex] = rs;

        _bufIndex = (_bufIndex + 1) % WindowSize;
        if (_bufCount < WindowSize)
            _bufCount++;
    }

    private float FilteredSensor(float[] buf, float currentRaw)
    {
        if (_bufCount <= 1)
            return currentRaw;

        int n = _bufCount;
        for (int i = 0; i < n; i++)
            _sortScratch[i] = buf[i];

        Array.Sort(_sortScratch, 0, n);
        float median = _sortScratch[n / 2];

        // If the latest sample is a sudden false 4.0 while the median shows a nearby obstacle,
        // clamp toward the median to avoid falsely assuming clear space.
        if (currentRaw >= SensorMax - 0.01f && median < 2.2f)
            return median;

        // Blend median and current sample for responsiveness + noise rejection
        return 0.65f * median + 0.35f * currentRaw;
    }

    private static float SanitizeSensor(float value)
    {
        if (float.IsNaN(value) || float.IsInfinity(value))
            return SensorMax;
        return Mathf.Clamp(value, 0.02f, SensorMax);
    }

    private static float NormalizeAngleDeg(float angle)
    {
        if (float.IsNaN(angle) || float.IsInfinity(angle))
            return 0.0f;

        while (angle > 180.0f)
            angle -= 360.0f;
        while (angle < -180.0f)
            angle += 360.0f;
        return angle;
    }

        float leftScore = left + 0.5f * leftSide + (goalAngle > 15.0f ? 0.25f : 0.0f);
        float rightScore = right + 0.5f * rightSide + (goalAngle < -15.0f ? 0.25f : 0.0f);

        if (Mathf.Abs(leftScore - rightScore) > 0.12f)
            return leftScore > rightScore;

        // Tie-breaker: turn toward goal sign
        return goalAngle >= 0.0f;
    }

    private int ChooseEscapeDirection(
        float goalAngle,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        float leftOpen = left + 0.6f * leftSide;
        float rightOpen = right + 0.6f * rightSide;

        if (Mathf.Abs(leftOpen - rightOpen) > 0.10f)
            return (leftOpen > rightOpen) ? 1 : -1;

        if (Mathf.Abs(goalAngle) > 20.0f)
            return (goalAngle >= 0.0f) ? 1 : -1;

        return (_totalTicks % 2 == 0) ? 1 : -1;
    }

    private (float speed01, float turn) ComputeSeekGoalCommand(
        float goalAngle,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        // Deadband around 0 deg prevents tiny left/right wiggle when aligned with the goal
        float effectiveAngle = (Mathf.Abs(goalAngle) < 4.0f) ? 0.0f : goalAngle;
        float goalTurn = effectiveAngle / 45.0f;

        float avoidanceBias = 0.0f;
        if (left < 0.50f)
            avoidanceBias -= (0.50f - left) * 1.3f;
        if (right < 0.50f)
            avoidanceBias += (0.50f - right) * 1.3f;
        if (leftSide < 0.24f)
            avoidanceBias -= (0.24f - leftSide) * 1.0f;
        if (rightSide < 0.24f)
            avoidanceBias += (0.24f - rightSide) * 1.0f;

        float turn = Mathf.Clamp(goalTurn + avoidanceBias, -1.0f, 1.0f);

        float angleFactor = Mathf.Clamp01(1.0f - Mathf.Abs(goalAngle) / 135.0f);
        float clearanceFactor = Mathf.Clamp01((front - 0.22f) / 1.10f);
        float speed01 = Mathf.Lerp(0.25f, 1.0f, angleFactor * clearanceFactor);

        if (Mathf.Abs(goalAngle) > 75.0f)
            speed01 = Mathf.Min(speed01, 0.35f);

        return (speed01, turn);
    }

    private (float speed01, float turn) ComputeWallFollowCommand(
        bool followLeftWall,
        float front,
        float diagAhead,
        float diagOpposite,
        float sideWall,
        float oppositeSideWall)
    {
        float towardWallSign = followLeftWall ? +1.0f : -1.0f;
        float awayWallSign = -towardWallSign;

        const float desiredSideDist = 0.35f;
        const float deadband = 0.04f;
        float turn = 0.0f;
        float speed01 = 0.68f;

        if (front < 0.56f || diagAhead < 0.46f)
        {
            float urgency = Mathf.Clamp01(1.0f - Mathf.Min(front / 0.56f, diagAhead / 0.46f));
            turn = awayWallSign * Mathf.Lerp(0.45f, 0.95f, urgency);
            speed01 = Mathf.Lerp(0.55f, 0.22f, urgency);
        }
        else if (sideWall < 1.30f)
        {
            float error = sideWall - desiredSideDist;
            if (Mathf.Abs(error) < deadband)
                error = 0.0f;
            turn = Mathf.Clamp(towardWallSign * (error * 1.8f), -0.55f, 0.55f);
            speed01 = 0.72f;
        }
        else
        {
            // Gentle curve back toward wall when outer corner is passed
            turn = towardWallSign * 0.32f;
            speed01 = 0.65f;
        }

        if (oppositeSideWall < 0.22f)
            turn += towardWallSign * ((0.22f - oppositeSideWall) * 1.2f);
        if (diagOpposite < 0.26f)
            turn += towardWallSign * ((0.26f - diagOpposite) * 1.1f);

        return (speed01, Mathf.Clamp(turn, -1.0f, 1.0f));
    }

    private (float speed01, float turn) ComputeEscapeCommand(float front, float left, float right)
    {
        float turn = _escapeSign * 0.95f;
        float speed01 = (front > 0.50f && left > 0.40f && right > 0.40f) ? 0.20f : 0.0f;
        return (speed01, turn);
    }

    private float ApplyProximitySpeedLimit(
        float speed01,
        float turn,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        float minAll = Mathf.Min(front, Mathf.Min(left, right));
        float minSide = Mathf.Min(leftSide, rightSide);

        if (minAll < 0.20f || minSide < 0.14f)
            return Mathf.Min(speed01, 0.12f);

        if (minAll < 0.34f || minSide < 0.20f)
            speed01 = Mathf.Min(speed01, 0.35f);
        else if (minAll < 0.55f)
            speed01 = Mathf.Min(speed01, 0.65f);

        float absTurn = Mathf.Abs(turn);
        if (absTurn > 0.75f)
            speed01 = Mathf.Min(speed01, 0.30f);
        else if (absTurn > 0.45f)
            speed01 = Mathf.Min(speed01, 0.55f);

        return speed01;
    }

    private bool IsGoalPathClear(
        float goalAngle,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        if (Mathf.Abs(goalAngle) <= 20.0f)
            return front > 0.75f && left > 0.52f && right > 0.52f;

        if (goalAngle > 20.0f && goalAngle <= 65.0f)
            return left > 0.75f && front > 0.52f;

        if (goalAngle < -20.0f && goalAngle >= -65.0f)
            return right > 0.75f && front > 0.52f;

        if (goalAngle > 65.0f && goalAngle <= 110.0f)
            return leftSide > 0.75f && left > 0.52f;

        if (goalAngle < -65.0f && goalAngle >= -110.0f)
            return rightSide > 0.75f && right > 0.52f;

        return false;
    }

    private void TrackSteeringOscillation(float turn)
    {
        int sign = 0;
        if (turn > 0.12f) sign = 1;
        else if (turn < -0.12f) sign = -1;

        if (sign != 0 && _prevTurnSign != 0 && sign != _prevTurnSign)
        {
            _oscillationScore += 2;
        }
        else if (_oscillationScore > 0)
        {
            _oscillationScore--;
        }

        if (sign != 0)
            _prevTurnSign = sign;
    }

    private void UpdateProgressTracking(float goalDist)
    {
        if (goalDist + 0.04f < _bestGoalDist)
        {
            _bestGoalDist = goalDist;
            _ticksWithoutProgress = 0;
            _oscillationScore = Mathf.Max(0, _oscillationScore - 2);
        }
        else
        {
            _ticksWithoutProgress++;
        }

        if (goalDist + 0.25f < _lastProgressGoalDist)
        {
            _lastProgressGoalDist = goalDist;
        }

        _prevGoalDist = goalDist;
    }

    private void PushSample(float f, float l, float r, float ls, float rs)
    {
        _frontBuf[_bufIndex] = f;
        _leftBuf[_bufIndex] = l;
        _rightBuf[_bufIndex] = r;
        _leftSideBuf[_bufIndex] = ls;
        _rightSideBuf[_bufIndex] = rs;

        _bufIndex = (_bufIndex + 1) % WindowSize;
        if (_bufCount < WindowSize)
            _bufCount++;
    }

    private float FilteredSensor(float[] buf, float currentRaw)
    {
        if (_bufCount <= 1)
            return currentRaw;

        int n = _bufCount;
        for (int i = 0; i < n; i++)
            _sortScratch[i] = buf[i];

        Array.Sort(_sortScratch, 0, n);
        float median = _sortScratch[n / 2];

        if (currentRaw >= SensorMax - 0.01f && median < 2.2f)
            return median;

        return 0.65f * median + 0.35f * currentRaw;
    }

    private static float SanitizeSensor(float value)
    {
        if (float.IsNaN(value) || float.IsInfinity(value))
            return SensorMax;
        return Mathf.Clamp(value, 0.02f, SensorMax);
    }

    private static float NormalizeAngleDeg(float angle)
    {
        if (float.IsNaN(angle) || float.IsInfinity(angle))
            return 0.0f;

        while (angle > 180.0f)
            angle -= 360.0f;
        while (angle < -180.0f)
            angle += 360.0f;
        return angle;
    }

Test report: 7/10 reached the goal, 0 collisions, 3 timeouts (got stuck), 0 runtime errors. Most common failure: gets stuck oscillating.

Even with the previous smoothing and dwell timers, oscillation usually happens in three places on 6 m × 6 m arenas with corridors and U-shapes:

Tug-of-war near obstacle edges: In SeekGoal, goalTurn and avoidanceBias can pull in opposite directions when approaching an obstacle at an angle, causing left/right wiggling or rapid SeekGoal ↔ FollowWall switching right on the threshold.

Corner / wall-loss limit cycles: When rounding a corner in FollowWall, sideWall crossing 1.30 m switches between tracking the wall and curving back toward it, which can cause a repeating weave if the sensor reading jitters near that threshold.

Repeated back-and-forth corrections when blocked: When progress stalls or steering alternates signs, brief or frequent direction changes can waste seconds without committing to a clear exit angle.

The updated LLM_Controller below tightens mode hysteresis (requiring consecutive clear ticks before returning to SeekGoal), smooths wall-loss/cornering arcs, and adds a committed sweep/escape lock when oscillation or stalled progress is detected.

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "LLM_Controller";

    private const float Dt = 0.05f;
    private const float MaxSpeed = 0.30f;       // m/s
    private const float MaxTurnRateDeg = 90.0f; // deg/s
    private const float SensorMax = 4.0f;
    private const int WindowSize = 5;

    private enum NavMode
    {
        SeekGoal,
        FollowWallLeft,
        FollowWallRight,
        CommittedTurn
    }

    private NavMode _mode;
    private int _modeTicks;
    private int _clearGoalTicks;
    private int _commitSign;          // +1 = left, -1 = right
    private int _commitTicksLeft;
    private int _wallFlipCount;
    private int _totalTicks;

    private float _bestGoalDist;
    private float _lastProgressGoalDist;
    private int _ticksWithoutProgress;
    private float _prevGoalDist;

    private float _prevTurnCmd;
    private int _prevTurnSign;
    private int _oscillationScore;

    private readonly float[] _frontBuf = new float[WindowSize];
    private readonly float[] _leftBuf = new float[WindowSize];
    private readonly float[] _rightBuf = new float[WindowSize];
    private readonly float[] _leftSideBuf = new float[WindowSize];
    private readonly float[] _rightSideBuf = new float[WindowSize];
    private readonly float[] _sortScratch = new float[WindowSize];
    private int _bufIndex;
    private int _bufCount;

    public void ResetController()
    {
        _mode = NavMode.SeekGoal;
        _modeTicks = 0;
        _clearGoalTicks = 0;
        _commitSign = 1;
        _commitTicksLeft = 0;
        _wallFlipCount = 0;
        _totalTicks = 0;

        _bestGoalDist = float.MaxValue;
        _lastProgressGoalDist = float.MaxValue;
        _ticksWithoutProgress = 0;
        _prevGoalDist = float.MaxValue;

        _prevTurnCmd = 0.0f;
        _prevTurnSign = 0;
        _oscillationScore = 0;

        _bufIndex = 0;
        _bufCount = 0;

        for (int i = 0; i < WindowSize; i++)
        {
            _frontBuf[i] = SensorMax;
            _leftBuf[i] = SensorMax;
            _rightBuf[i] = SensorMax;
            _leftSideBuf[i] = SensorMax;
            _rightSideBuf[i] = SensorMax;
        }
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        _totalTicks++;
        _modeTicks++;

        float rawFront = SanitizeSensor(obs.front);
        float rawLeft = SanitizeSensor(obs.left);
        float rawRight = SanitizeSensor(obs.right);
        float rawLeftSide = SanitizeSensor(obs.leftSide);
        float rawRightSide = SanitizeSensor(obs.rightSide);

        PushSample(rawFront, rawLeft, rawRight, rawLeftSide, rawRightSide);

        float front = FilteredSensor(_frontBuf, rawFront);
        float left = FilteredSensor(_leftBuf, rawLeft);
        float right = FilteredSensor(_rightBuf, rawRight);
        float leftSide = FilteredSensor(_leftSideBuf, rawLeftSide);
        float rightSide = FilteredSensor(_rightSideBuf, rawRightSide);

        float goalDist = Mathf.Max(0.0f, obs.goalDistance);
        float goalAngle = NormalizeAngleDeg(obs.goalAngleDeg);

        UpdateProgressTracking(goalDist);

        if (goalDist < 0.10f)
        {
            float settleTurn = Mathf.Clamp(goalAngle / 40.0f, -0.35f, 0.35f);
            float smoothedSettle = Mathf.Lerp(_prevTurnCmd, settleTurn, 0.5f);
            _prevTurnCmd = smoothedSettle;
            return (0.15f, smoothedSettle);
        }

        float minForwardCone = Mathf.Min(front, Mathf.Min(left, right));
        bool blockedAhead = front < 0.46f || minForwardCone < 0.32f;
        bool tightPocket = blockedAhead && left < 0.58f && right < 0.58f;

        // Track consecutive ticks where the direct line toward the goal looks open
        if (IsGoalPathClear(goalAngle, front, left, right, leftSide, rightSide))
            _clearGoalTicks++;
        else
            _clearGoalTicks = 0;

        // State transitions with hysteresis to eliminate chatter and oscillation
        if (_mode == NavMode.CommittedTurn)
        {
            _commitTicksLeft--;

            bool pathOpenedEarly = _modeTicks >= 20 &&
                                   front > 0.85f &&
                                   left > 0.62f &&
                                   right > 0.62f;

            if (pathOpenedEarly || _commitTicksLeft <= 0)
            {
                // Transition smoothly into wall-following on the opposite side of the turn
                // so we hug the obstacle boundary rather than swinging back into it
                _mode = (_commitSign > 0) ? NavMode.FollowWallRight : NavMode.FollowWallLeft;
                _modeTicks = 0;
                _clearGoalTicks = 0;
                _ticksWithoutProgress = 0;
                _lastProgressGoalDist = goalDist;
                _oscillationScore = 0;
            }
        }
        else if (tightPocket || (_oscillationScore >= 7 && _ticksWithoutProgress >= 30))
        {
            StartCommittedTurn(
                sign: ChooseCommittedDirection(goalAngle, left, right, leftSide, rightSide),
                ticks: tightPocket ? 46 : 36
            );
        }
        else if (_mode == NavMode.SeekGoal)
        {
            if (blockedAhead && _modeTicks >= 8)
            {
                bool followLeft = ChooseWallFollowLeft(goalAngle, left, right, leftSide, rightSide);
                _mode = followLeft ? NavMode.FollowWallLeft : NavMode.FollowWallRight;
                _modeTicks = 0;
                _clearGoalTicks = 0;
            }
        }
        else // FollowWallLeft or FollowWallRight
        {
            // Require both a minimum dwell time and sustained clear path before resuming SeekGoal
            bool minDwellMet = _modeTicks >= 60;
            bool sustainedClear = _clearGoalTicks >= 8;
            bool progressed = goalDist + 0.16f < _lastProgressGoalDist;

            if (minDwellMet && sustainedClear && !blockedAhead && (progressed || _modeTicks >= 95))
            {
                _mode = NavMode.SeekGoal;
                _modeTicks = 0;
                _clearGoalTicks = 0;
                _wallFlipCount = 0;
            }
            else if (_ticksWithoutProgress >= 140)
            {
                if (_wallFlipCount < 1)
                {
                    _wallFlipCount++;
                    _mode = (_mode == NavMode.FollowWallLeft) ? NavMode.FollowWallRight : NavMode.FollowWallLeft;
                    _modeTicks = 0;
                    _clearGoalTicks = 0;
                    _ticksWithoutProgress = 0;
                    _lastProgressGoalDist = goalDist;
                    _oscillationScore = 0;
                }
                else
                {
                    _wallFlipCount = 0;
                    StartCommittedTurn(
                        sign: ChooseCommittedDirection(goalAngle, left, right, leftSide, rightSide),
                        ticks: 48
                    );
                }
            }
        }

        float speed01;
        float rawTurn;

        switch (_mode)
        {
            case NavMode.CommittedTurn:
                (speed01, rawTurn) = ComputeCommittedTurnCommand(front, left, right);
                break;

            case NavMode.FollowWallLeft:
                (speed01, rawTurn) = ComputeWallFollowCommand(
                    followLeftWall: true,
                    front: front,
                    diagAhead: left,
                    diagOpposite: right,
                    sideWall: leftSide,
                    oppositeSideWall: rightSide
                );
                break;

            case NavMode.FollowWallRight:
                (speed01, rawTurn) = ComputeWallFollowCommand(
                    followLeftWall: false,
                    front: front,
                    diagAhead: right,
                    diagOpposite: left,
                    sideWall: rightSide,
                    oppositeSideWall: leftSide
                );
                break;

            case NavMode.SeekGoal:
            default:
                (speed01, rawTurn) = ComputeSeekGoalCommand(goalAngle, front, left, right, leftSide, rightSide);
                break;
        }

        // Low-pass filter on turn command dampens high-frequency sensor noise
        float alpha = (_mode == NavMode.CommittedTurn) ? 0.75f : 0.45f;
        float turn = Mathf.Lerp(_prevTurnCmd, rawTurn, alpha);

        TrackSteeringOscillation(turn);
        _prevTurnCmd = turn;

        speed01 = ApplyProximitySpeedLimit(speed01, turn, front, left, right, leftSide, rightSide);
        turn = Mathf.Clamp(turn, -1.0f, 1.0f);
        speed01 = Mathf.Clamp(speed01, 0.0f, 1.0f);

        return (speed01, turn);
    }

The drop to 5/10 with timeouts caused by oscillation points to a few specific feedback loops in the previous logic:

Front/side tug-of-war in wall-following: Switching abruptly between turning away from front/diagAhead and turning toward sideWall whenever sideWall < 1.45 m creates a limit cycle right at the obstacle corner, causing the rover to weave or pivot back and forth instead of gliding past the edge.

False-positive oscillation triggers: Flipping turn sign during normal obstacle avoidance or cornering is expected; treating that as "oscillation" and entering a long CommittedTurn can swing the rover off-line or into another wall, wasting time.

Premature/late exit from wall-following: Without requiring actual progress past the obstacle (goalDist improving relative to when wall-following started), the rover can re-enter SeekGoal facing the same obstacle or stay stuck on a loop.

Below is an improved LLM_Controller that uses a 7-sample median + EMA filter for noisier/false-4.0 readings, a continuous damped wall-following law with a deadband and gentle cornering curve, a progress-gated leave condition, and a clean unstick maneuver when no new best goal distance is reached over time.

    private (float speed01, float turn) ComputeSeekGoal(
        float goalAngle,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        float angledGoal = (Mathf.Abs(goalAngle) < 3.5f) ? 0.0f : goalAngle;
        float turn = angledGoal / 45.0f;

        if (left < 0.45f)
            turn -= (0.45f - left) * 1.15f;
        if (right < 0.45f)
            turn += (0.45f - right) * 1.15f;
        if (leftSide < 0.22f)
            turn -= (0.22f - leftSide) * 0.9f;
        if (rightSide < 0.22f)
            turn += (0.22f - rightSide) * 0.9f;

        float align = Mathf.Clamp01(1.0f - Mathf.Abs(goalAngle) / 120.0f);
        float clear = Mathf.Clamp01((front - 0.24f) / 1.0f);
        float speed01 = Mathf.Lerp(0.25f, 0.95f, align * clear);

        if (Mathf.Abs(goalAngle) > 70.0f)
            speed01 = Mathf.Min(speed01, 0.35f);

        return (speed01, Mathf.Clamp(turn, -1.0f, 1.0f));
    }

    private (float speed01, float turn) ComputeSmoothWallFollow(
        bool followLeft,
        float front,
        float diagSame,
        float diagOpp,
        float sideSame,
        float sideOpp)
    {
        // +1 = left, -1 = right
        float towardWall = followLeft ? +1.0f : -1.0f;
        float awayWall = -towardWall;

        const float targetSideDist = 0.34f;
        const float deadband = 0.03f;

        float turn;
        float speed01;

        // Blended obstacle clearance + wall distance so there is no abrupt mode flip at corners
        float frontThreat = Mathf.Clamp01((0.55f - front) / 0.38f);
        float diagThreat = Mathf.Clamp01((0.44f - diagSame) / 0.28f);
        float obstacleAvoid = Mathf.Max(frontThreat, diagThreat);

        if (obstacleAvoid > 0.01f)
        {
            turn = awayWall * Mathf.Lerp(0.35f, 0.95f, obstacleAvoid);
            speed01 = Mathf.Lerp(0.60f, 0.24f, obstacleAvoid);
        }
        else if (sideSame < 1.05f)
        {
            float err = sideSame - targetSideDist;
            if (Mathf.Abs(err) < deadband)
                err = 0.0f;
            turn = Mathf.Clamp(towardWall * (err * 1.4f), -0.35f, 0.35f);
            speed01 = 0.68f;
        }
        else
        {
            // Passed an outer corner: curve gently toward the wall side to re-acquire it
            turn = towardWall * 0.26f;
            speed01 = 0.62f;
        }

        // Prevent brushing walls/obstacles on the opposite side (e.g., narrow corridors)
        if (sideOpp < 0.22f)
            turn += towardWall * ((0.22f - sideOpp) * 1.1f);
        if (diagOpp < 0.26f)
            turn += towardWall * ((0.26f - diagOpp) * 1.0f);

        return (speed01, Mathf.Clamp(turn, -1.0f, 1.0f));
    }

    private float ApplySafetySpeedClamp(
        float speed01,
        float turn,
        float front,
        float left,
        float right,
        float leftSide,
        float rightSide)
    {
        float minFwd = Mathf.Min(front, Mathf.Min(left, right));
        float minSide = Mathf.Min(leftSide, rightSide);

        if (minFwd < 0.20f || minSide < 0.14f)
            return Mathf.Min(speed01, 0.15f);

        if (minFwd < 0.32f || minSide < 0.19f)
            speed01 = Mathf.Min(speed01, 0.35f);
        else if (minFwd < 0.50f)
            speed01 = Mathf.Min(speed01, 0.60f);

        if (Mathf.Abs(turn) > 0.70f)
            speed01 = Mathf.Min(speed01, 0.32f);

        return speed01;
    }

    private bool ChooseWallFollowLeft(

        float rightRoom = right + 0.5f * rightSide;

        return goalAngle >= 0.0f;
