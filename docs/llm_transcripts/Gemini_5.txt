    public (float speed01, float turn) Decide(Observation obs)
    {
        _tick++;
        _modeTicks++;

        // Track goal progress for escape / re-evaluation heuristics
        if (obs.goalDistance < _bestGoalDist - 0.05f)
        {
            _bestGoalDist = obs.goalDistance;
            _ticksSinceProgress = 0;
        }
        else
        {
            _ticksSinceProgress++;
        }

        // Filter raw ultrasonic sensors against Gaussian noise and occasional false 4.0m readings
        float f = FilterSensor(0, obs.front);
        float l45 = FilterSensor(1, obs.left);
        float r45 = FilterSensor(2, obs.right);
        float l90 = FilterSensor(3, obs.leftSide);
        float r90 = FilterSensor(4, obs.rightSide);

        // Effective front & diagonal hazard distances (rover is 0.20m long x 0.15m wide)
        float minFrontCone = Mathf.Min(f, Mathf.Min(l45 * 1.08f, r45 * 1.08f));
        float minAnyClose = Mathf.Min(minFrontCone, Mathf.Min(l90 * 1.15f, r90 * 1.15f));

        // Goal reached or very close: approach smoothly
        if (obs.goalDistance < 0.12f && minFrontCone > 0.22f)
        {
            float tGoal = Mathf.Clamp(obs.goalAngleDeg / 35.0f, -1f, 1f);
            float sGoal = Mathf.Clamp01(0.45f * (1f - Mathf.Abs(tGoal) * 0.5f));
            return ApplySmoothing(sGoal, tGoal);
        }

        // Mode transitions
        bool frontBlocked = f < 0.42f || l45 < 0.30f || r45 < 0.30f;
        bool emergencyClose = f < 0.22f || l45 < 0.16f || r45 < 0.16f || minAnyClose < 0.13f;

        if (_mode == NavMode.SeekGoal)
        {
            if (emergencyClose)
            {
                EnterWallFollow(ChooseWallSide(obs.goalAngleDeg, l45, r45, l90, r90), true);
            }
            else if (frontBlocked)
            {
                EnterWallFollow(ChooseWallSide(obs.goalAngleDeg, l45, r45, l90, r90), false);
            }
        }
        else if (_mode == NavMode.PivotClear)
        {
            // Pivot until front/diagonal cone has enough room to move or wall-follow
            if (f > 0.48f && l45 > 0.34f && r45 > 0.34f && _modeTicks >= 4)
            {
                _mode = NavMode.SeekGoal;
                _modeTicks = 0;
            }
            else if (_modeTicks > 45) // ~2.25s max pivot before switching to wall follow
            {
                EnterWallFollow(ChooseWallSide(obs.goalAngleDeg, l45, r45, l90, r90), false);
            }
        }
        else if (_mode == NavMode.FollowLeftWall || _mode == NavMode.FollowRightWall)
        {
            bool leftWall = (_mode == NavMode.FollowLeftWall);
            float sideDist = leftWall ? l90 : r90;
            float diagWallDist = leftWall ? l45 : r45;
            float oppDiagDist = leftWall ? r45 : l45;

            // Check if we can leave the wall toward a clear goal direction
            float goalErr = Mathf.Abs(obs.goalAngleDeg);
            bool goalRoughlyAhead = goalErr < 55.0f;
            bool pathTowardGoalClear =

            // Leave wall if path toward goal is clear after following obstacle a bit, or if we lost the wall
            bool lostWall = sideDist > 1.25f && diagWallDist > 1.15f && _modeTicks > 8;
            bool canRejoinGoal = goalRoughlyAhead && pathTowardGoalClear && _modeTicks > 12 && oppDiagDist > 0.35f;

            // If stuck in a tight U-turn / dead-end for too long without goal progress, switch wall side
            if (_ticksSinceProgress > 220 && _modeTicks > 40) // ~11s without progress
            {
                _ticksSinceProgress = 100;
                EnterWallFollow(leftWall ? NavMode.FollowRightWall : NavMode.FollowLeftWall, frontBlocked);
            }
            else if (lostWall || canRejoinGoal)
            {
                _mode = NavMode.SeekGoal;
                _modeTicks = 0;
            }
        }

        float speed01;
        float turn;

        switch (_mode)
        {
            case NavMode.PivotClear:
                {
                    // Turn toward the more open side (or away from closer obstacle) with minimal forward creep
                    float leftOpen = l45 + 0.7f * l90;
                    float rightOpen = r45 + 0.7f * r90;
                    float dir = (leftOpen >= rightOpen) ? 1.0f : -1.0f;
                    if (Mathf.Abs(leftOpen - rightOpen) < 0.08f)
                    {
                        dir = (obs.goalAngleDeg >= 0f) ? 1.0f : -1.0f;
                    }
                    turn = dir * 0.85f;
                    speed01 = (f > 0.24f && l45 > 0.18f && r45 > 0.18f) ? 0.12f : 0.0f;
                    break;
                }

            case NavMode.FollowLeftWall:
                {
                    ComputeWallFollow(
                        followLeft: true,
                        f: f,
                        diagSame: l45,
                        diagOpp: r45,
                        sideSame: l90,
                        sideOpp: r90,
                        goalAngleDeg: obs.goalAngleDeg,
                        out speed01,
                        out turn
                    );
                    break;
                }

            case NavMode.FollowRightWall:
                {
                    ComputeWallFollow(
                        followLeft: false,
                        f: f,
                        diagSame: r45,
                        diagOpp: l45,
                        sideSame: r90,
                        sideOpp: l90,
                        goalAngleDeg: obs.goalAngleDeg,
                        out speed01,
                        out turn
                    );
                    break;
                }

            case NavMode.SeekGoal:
            default:
                {
                    // Base turn toward goal
                    float desiredTurn = Mathf.Clamp(obs.goalAngleDeg / 38.0f, -1.0f, 1.0f);

                    // Repulsive steering bias from nearby obstacles (positive turn = steer left, negative = right)
                    float leftThreat = ThreatWeight(l45, 0.55f) * 1.15f + ThreatWeight(l90, 0.34f) * 0.75f;
                    float rightThreat = ThreatWeight(r45, 0.55f) * 1.15f + ThreatWeight(r90, 0.34f) * 0.75f;
                    float frontThreat = ThreatWeight(f, 0.65f);

                    float avoidBias = (rightThreat - leftThreat);
                    if (frontThreat > 0.05f)
                    {
                        float fSign = (obs.goalAngleDeg >= 0f) ? 1.0f : -1.0f;
                        if (Mathf.Abs(leftThreat - rightThreat) > 0.15f)
                        {
                            fSign = (rightThreat > leftThreat) ? 1.0f : -1.0f;
                        }
                        avoidBias += fSign * frontThreat * 1.25f;
                    }

                    turn = Mathf.Clamp(desiredTurn * (1.0f - 0.55f * Mathf.Clamp01(frontThreat)) + avoidBias, -1.0f, 1.0f);

                    // Speed modulation: slow down for sharp turns or close obstacles
                    float turnFactor = 1.0f - 0.55f * Mathf.Clamp01(Mathf.Abs(obs.goalAngleDeg) / 90.0f);
                    float distFactor = Mathf.Clamp01((minFrontCone - 0.18f) / 0.65f);
                    speed01 = Mathf.Clamp(0.95f * turnFactor * (0.25f + 0.75f * distFactor), 0.12f, 1.0f);
                    break;
                }
        }

        return ApplySmoothing(speed01, turn);
    }

    private NavMode ChooseWallSide(float goalAngleDeg, float l45, float r45, float l90, float r90)
    {
        // Prefer following the wall on the side opposite the goal so turning toward the goal opens up naturally,
        // unless one side is much closer/clearer.
        float leftScore = l45 + 0.6f * l90 + (goalAngleDeg < -15f ? 0.25f : -0.10f);
        float rightScore = r45 + 0.6f * r90 + (goalAngleDeg > 15f ? 0.25f : -0.10f);

        // In tight spaces, hug whichever side has a clearer wall within reach
        if (l90 < 0.45f && r90 > 0.85f) return NavMode.FollowLeftWall;
        if (r90 < 0.45f && l90 > 0.85f) return NavMode.FollowRightWall;

        return (leftScore >= rightScore) ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
    }

    private void ComputeWallFollow(
        bool followLeft,
        float f,
        float diagSame,
        float diagOpp,
        float sideSame,
        float sideOpp,
        float goalAngleDeg,
        out float speed01,
        out float turn)
    {
        // Sign helper: +1 means turn left, -1 means turn right
        float s = followLeft ? 1.0f : -1.0f;

        // Target distance to keep from the followed side wall (rover half-width is 0.075m)
        const float targetSideDist = 0.24f;

        // Inside corner / obstacle directly ahead: turn sharply away from followed wall
        if (f < 0.30f || diagSame < 0.24f)
        {
            turn = -s * 0.95f;
            speed01 = (f < 0.18f || diagSame < 0.15f) ? 0.05f : 0.22f;
            return;
        }

        // Corridor centering if both sides are narrow
        if (sideSame < 0.50f && sideOpp < 0.50f)
        {
            float centerErr = (sideOpp - sideSame); // positive => more room on opposite side
            float corrTurn = Mathf.Clamp(centerErr * 2.2f, -0.65f, 0.65f);
            float fSlow = Mathf.Clamp01((f - 0.25f) / 0.60f);
            turn = corrTurn;
            speed01 = Mathf.Clamp(0.55f * fSlow, 0.15f, 0.65f);
            return;
        }

        // Standard wall following PD-like control:
        // If sideSame < targetSideDist -> too close to wall -> steer away (-s)
        // If sideSame > targetSideDist -> drifting away from wall -> steer slightly toward wall (+s)
        float distError = sideSame - targetSideDist;
        float wallTurn = s * Mathf.Clamp(distError * 2.4f, -0.55f, 0.55f);

        // Also react to diagonal sensor on the wall side so we don't clip protruding corners
        if (diagSame < 0.42f)
        {
            float diagPush = (0.42f - diagSame) * 2.5f;
            wallTurn -= s * Mathf.Clamp(diagPush, 0f, 0.75f);
        }

        // Gentle nudging toward goal if turning toward goal moves away from the wall
        float goalNudge = Mathf.Clamp(goalAngleDeg / 120.0f, -0.2f, 0.2f);
        if (s * goalNudge < 0f && sideSame < 0.35f)
        {
            goalNudge = 0f;
        }

        turn = Mathf.Clamp(wallTurn + goalNudge, -1.0f, 1.0f);

        float minClear = Mathf.Min(f, Mathf.Min(diagSame, Mathf.Min(diagOpp, sideSame)));
        float clearFactor = Mathf.Clamp01((minClear - 0.16f) / 0.55f);
        float turnPenalty = 1.0f - 0.45f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.80f * clearFactor * turnPenalty, 0.15f, 0.85f);
    }

    private static float ThreatWeight(float dist, float influenceRadius)
    {
        if (dist >= influenceRadius) return 0f;
        float clamped = Mathf.Max(dist, 0.02f);
        float t = (influenceRadius - clamped) / influenceRadius;
        return t * t;
    }

    private float FilterSensor(int sensorIndex, float raw)
    {
        float clamped = Mathf.Clamp(raw, 0.02f, SensorMaxRange);

        // Detect likely false 4.0m reading when recent readings were significantly closer
        if (clamped >= SensorMaxRange - 1e-4f && _histCount[sensorIndex] > 0)
        {
            float recentAvg = ComputeAverage(sensorIndex);
            if (recentAvg < 1.35f)
            {
                // Blend a conservative reading so an occasional false 4.0 doesn't cause a sudden lunge
                clamped = Mathf.Clamp(recentAvg + 0.15f, 0.02f, 1.8f);
            }
        }

        int idx = _histIdx[sensorIndex];
        _hist[sensorIndex, idx] = clamped;
        _histIdx[sensorIndex] = (idx + 1) % HistorySize;
        if (_histCount[sensorIndex] < HistorySize)
        {
            _histCount[sensorIndex]++;
        }

        // Use median of recent window to reject single-sample noise spikes
        int count = _histCount[sensorIndex];
        float[] temp = new float[count];
        for (int i = 0; i < count; i++)
        {
            temp[i] = _hist[sensorIndex, i];
        }
        Array.Sort(temp);
        float median = temp[count / 2];

        // Combine median and mean slightly for smooth, responsive distance estimate
        float mean = ComputeAverage(sensorIndex);
        return Mathf.Clamp(0.65f * median + 0.35f * mean, 0.02f, SensorMaxRange);
    }

    private float ComputeAverage(int sensorIndex)
    {
        int count = _histCount[sensorIndex];
        if (count == 0) return SensorMaxRange;
        float sum = 0f;
        for (int i = 0; i < count; i++)
        {
            sum += _hist[sensorIndex, i];
        }
        return sum / count;
    }

    private (float speed01, float turn) ApplySmoothing(float targetSpeed, float targetTurn)
    {
        // Limit per-tick changes slightly for smooth actuation while staying responsive
        float maxTurnDelta = 0.65f;
        float turn = Mathf.Clamp(targetTurn, _prevTurn - maxTurnDelta, _prevTurn + maxTurnDelta);
        turn = Mathf.Clamp(turn, -1.0f, 1.0f);

        float maxSpeedDelta = 0.35f;
        float speed = Mathf.Clamp(targetSpeed, _prevSpeed - maxSpeedDelta, _prevSpeed + maxSpeedDelta);
        speed = Mathf.Clamp01(speed);

        _prevTurn = turn;
        _prevSpeed = speed;
        return (speed, turn);
    }

Test report: 9/10 reached the goal, 0 collisions, 1 timeouts (got stuck), 0 runtime errors. Most common failure: gets stuck oscillating.

Getting 9/10 with 0 collisions is a strong start, and oscillation usually comes down to three specific culprits in reactive controllers:

Dropped wall-follow target after pivoting: EnterWallFollow(..., forcePivotFirst: true) switched into PivotClear without saving the intended wall-follow side, so after clearing the front cone it returned to SeekGoal, hit the same obstacle again, and looped.

Mode & sign ping-ponging near thresholds: Rapidly crossing frontBlocked or canRejoinGoal conditions—or flipping between left/right avoidance when leftThreat and rightThreat are nearly tied—causes back-and-forth turn reversals.

High proportional gains without damping: Sharp gain steps around targetSideDist or centerErr can overshoot and oscillate before settling.

Key Fixes in This Version
Explicit pending wall mode + hysteresis/cooldowns: Stores _pendingWallMode during PivotClear and enforces minimum dwell ticks (_modeTicks) plus asymmetric enter/exit thresholds (canRejoinGoal requires clearer space and slightly longer dwell) so modes don't chatter.

Sticky avoidance / wall-follow bias: Breaks near-ties consistently using a persistent preference (_preferredSide) rather than flipping sign tick-to-tick, and only switches wall sides after a longer no-progress window.

Turn-rate & reversal damping: Adds subtle derivative/reversal damping on turn and tighter per-tick slew limits so the rover settles smoothly along walls, corridors, and U-turns without thrashing.

                {
                    float desiredTurn = Mathf.Clamp(obs.goalAngleDeg / 42.0f, -1.0f, 1.0f);

                    float leftThreat = ThreatWeight(l45, 0.52f) * 1.05f + ThreatWeight(l90, 0.32f) * 0.65f;
                    float rightThreat = ThreatWeight(r45, 0.52f) * 1.05f + ThreatWeight(r90, 0.32f) * 0.65f;
                    float frontThreat = ThreatWeight(f, 0.62f);

                    // Update sticky turn preference only when there is a clear asymmetry
                    float threatDiff = rightThreat - leftThreat;
                    if (Mathf.Abs(threatDiff) > 0.14f)
                    {
                        _preferredTurnSign = threatDiff > 0f ? 1 : -1;
                    }
                    else if (Mathf.Abs(obs.goalAngleDeg) > 20f)
                    {
                        _preferredTurnSign = obs.goalAngleDeg >= 0f ? 1 : -1;
                    }

                    float avoidBias = Mathf.Clamp(threatDiff * 0.85f, -0.65f, 0.65f);
                    if (frontThreat > 0.08f)
                    {
                        avoidBias += _preferredTurnSign * frontThreat * 0.85f;
                    }

                    float rawTurn = desiredTurn * (1.0f - 0.50f * Mathf.Clamp01(frontThreat)) + avoidBias;
                    // Dampen rapid sign reversals around zero
                    if (Mathf.Sign(rawTurn) != Mathf.Sign(_prevTurn) && Mathf.Abs(_prevTurn) > 0.15f)
                    {
                        rawTurn *= 0.65f;
                    }

                    turn = Mathf.Clamp(rawTurn, -1.0f, 1.0f);

                    float turnFactor = 1.0f - 0.50f * Mathf.Clamp01(Mathf.Abs(turn));
                    float distFactor = Mathf.Clamp01((minFrontCone - 0.18f) / 0.65f);
                    speed01 = Mathf.Clamp(0.88f * turnFactor * (0.25f + 0.75f * distFactor), 0.14f, 0.95f);

                    return ApplySmoothing(speed01, turn, maxTurnDelta: 0.32f, maxSpeedDelta: 0.22f);
                }
        }
    }

    private void EnterWallFollow(NavMode targetFollowMode, bool pivotFirst)
    {
        _modeTicks = 0;
        _pendingWallMode = targetFollowMode;
        _preferredTurnSign = (targetFollowMode == NavMode.FollowLeftWall) ? 1 : -1;
        _mode = pivotFirst ? NavMode.PivotClear : targetFollowMode;
    }

    private NavMode ChooseWallSide(float goalAngleDeg, float l45, float r45, float l90, float r90)
    {
        if (l90 < 0.45f && r90 > 0.85f) return NavMode.FollowLeftWall;
        if (r90 < 0.45f && l90 > 0.85f) return NavMode.FollowRightWall;

        float leftScore = l45 + 0.55f * l90 + (goalAngleDeg < -20f ? 0.22f : -0.08f);
        float rightScore = r45 + 0.55f * r90 + (goalAngleDeg > 20f ? 0.22f : -0.08f);

        if (Mathf.Abs(leftScore - rightScore) < 0.08f)
        {
            return _preferredTurnSign >= 0 ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
        }

        return (leftScore > rightScore) ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
    }

    private void ComputeWallFollow(
        bool followLeft,
        float f,
        float diagSame,
        float diagOpp,
        float sideSame,
        float sideOpp,
        float goalAngleDeg,
        out float speed01,
        out float turn)
    {
        float s = followLeft ? 1.0f : -1.0f;
        const float targetSideDist = 0.25f;

        if (f < 0.29f || diagSame < 0.23f)
        {
            turn = -s * 0.85f;
            speed01 = (f < 0.18f || diagSame < 0.15f) ? 0.05f : 0.20f;
            return;
        }

        // Narrow corridor centering with deadband to avoid left-right chatter
        if (sideSame < 0.48f && sideOpp < 0.48f)
        {
            float centerErr = sideOpp - sideSame;
            float corrTurn = Mathf.Abs(centerErr) < 0.03f ? 0f : Mathf.Clamp(centerErr * 1.6f, -0.45f, 0.45f);
            float fSlow = Mathf.Clamp01((f - 0.24f) / 0.60f);
            turn = corrTurn;
            speed01 = Mathf.Clamp(0.50f * fSlow, 0.15f, 0.60f);
            return;
        }

        float distError = sideSame - targetSideDist;
        float deadbandError = Mathf.Abs(distError) < 0.025f ? 0f : distError;
        float wallTurn = s * Mathf.Clamp(deadbandError * 1.75f, -0.42f, 0.42f);

        if (diagSame < 0.40f)
        {
            float diagPush = (0.40f - diagSame) * 1.9f;
            wallTurn -= s * Mathf.Clamp(diagPush, 0f, 0.60f);
        }

        float goalNudge = Mathf.Clamp(goalAngleDeg / 140.0f, -0.15f, 0.15f);
        if (s * goalNudge < 0f && sideSame < 0.36f)
        {
            goalNudge = 0f;
        }

        turn = Mathf.Clamp(wallTurn + goalNudge, -1.0f, 1.0f);

        float minClear = Mathf.Min(f, Mathf.Min(diagSame, Mathf.Min(diagOpp, sideSame)));
        float clearFactor = Mathf.Clamp01((minClear - 0.16f) / 0.55f);
        float turnPenalty = 1.0f - 0.40f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.75f * clearFactor * turnPenalty, 0.15f, 0.80f);
    }

    private static float ThreatWeight(float dist, float influenceRadius)
    {
        if (dist >= influenceRadius) return 0f;
        float clamped = Mathf.Max(dist, 0.02f);
        float t = (influenceRadius - clamped) / influenceRadius;
        return t * t;
    }

    private float FilterSensor(int sensorIndex, float raw)
    {
        float clamped = Mathf.Clamp(raw, 0.02f, MaxSensorRange);

        if (clamped >= MaxSensorRange - 1e-4f && _histCount[sensorIndex] > 0)
        {
            float recentAvg = ComputeAverage(sensorIndex);
            if (recentAvg < 1.35f)
            {
                clamped = Mathf.Clamp(recentAvg + 0.15f, 0.02f, 1.8f);
            }
        }

        int idx = _histIdx[sensorIndex];
        _hist[sensorIndex, idx] = clamped;
        _histIdx[sensorIndex] = (idx + 1) % HistorySize;
        if (_histCount[sensorIndex] < HistorySize)
        {
            _histCount[sensorIndex]++;
        }

        int count = _histCount[sensorIndex];
        float[] temp = new float[count];
        for (int i = 0; i < count; i++)
        {
            temp[i] = _hist[sensorIndex, i];
        }
        Array.Sort(temp);
        float median = temp[count / 2];

        float mean = ComputeAverage(sensorIndex);
        return Mathf.Clamp(0.70f * median + 0.30f * mean, 0.02f, MaxSensorRange);
    }

    private float ComputeAverage(int sensorIndex)
    {
        int count = _histCount[sensorIndex];
        if (count == 0) return MaxSensorRange;
        float sum = 0f;
        for (int i = 0; i < count; i++)
        {
            sum += _hist[sensorIndex, i];
        }
        return sum / count;
    }

    private (float speed01, float turn) ApplySmoothing(
        float targetSpeed,
        float targetTurn,
        float maxTurnDelta,
        float maxSpeedDelta)
    {
        float turn = Mathf.Clamp(targetTurn, _prevTurn - maxTurnDelta, _prevTurn + maxTurnDelta);
        turn = Mathf.Clamp(turn, -1.0f, 1.0f);

        float speed = Mathf.Clamp(targetSpeed, _prevSpeed - maxSpeedDelta, _prevSpeed + maxSpeedDelta);
        speed = Mathf.Clamp01(speed);

        _prevTurn = turn;
        _prevSpeed = speed;
        return (speed, turn);
    }

Stricter exit from wall-following: Only returns to SeekGoal when the path in the goal direction is genuinely open (f > 0.95f and the goal-side diagonal is clear) and it has spent enough ticks rounding the obstacle, preventing SeekGoal ↔ FollowWall ping-ponging.

using System;
using UnityEngine;

public sealed class LLM_Controller : IRoverController
{
    public string Name => "LLM_StableWallGoalController";

    private enum NavMode
    {
        SeekGoal,
        FollowLeftWall,
        FollowRightWall,
        EscapeTurn
    }

    private const float MaxSensorRange = 4.0f;
    private const int HistorySize = 5;

    private NavMode _mode;
    private int _modeTicks;
    private int _escapeTicks;
    private float _escapeDir; // +1 left, -1 right
    private int _wallSwitchCount;

    private float _bestGoalDist;
    private int _ticksSinceProgress;

    private float _prevRawTurn;
    private int _signFlips;
    private int _flipWindowTicks;

    private float[,] _hist;
    private int[] _histCount;
    private int[] _histIdx;

    public LLM_Controller()
    {
        _hist = new float[5, HistorySize];
        _histCount = new int[5];
        _histIdx = new int[5];
        ResetController();
    }

    public void ResetController()
    {
        _mode = NavMode.SeekGoal;
        _modeTicks = 0;
        _escapeTicks = 0;
        _escapeDir = 1.0f;
        _wallSwitchCount = 0;

        _bestGoalDist = float.MaxValue;
        _ticksSinceProgress = 0;

        _prevRawTurn = 0f;
        _signFlips = 0;
        _flipWindowTicks = 0;

        for (int s = 0; s < 5; s++)
        {
            _histCount[s] = 0;
            _histIdx[s] = 0;
            for (int k = 0; k < HistorySize; k++)
            {
                _hist[s, k] = MaxSensorRange;
            }
        }
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        _modeTicks++;

        if (obs.goalDistance < _bestGoalDist - 0.04f)
        {
            _bestGoalDist = obs.goalDistance;
            _ticksSinceProgress = 0;
        }
        else
        {
            _ticksSinceProgress++;
        }

        float f = FilterSensor(0, obs.front);
        float l45 = FilterSensor(1, obs.left);
        float r45 = FilterSensor(2, obs.right);
        float l90 = FilterSensor(3, obs.leftSide);
        float r90 = FilterSensor(4, obs.rightSide);

        float minFront = Mathf.Min(f, Mathf.Min(l45, r45));

        // Goal close and front clear: drive smoothly into goal
        if (obs.goalDistance < 0.14f && minFront > 0.24f)
        {
            float tGoal = Mathf.Clamp(obs.goalAngleDeg / 32.0f, -1f, 1f);
            float sGoal = Mathf.Clamp01(0.45f * (1f - 0.4f * Mathf.Abs(tGoal)));
            RecordTurnSign(tGoal);
            return (sGoal, tGoal);
        }

        // Detect sustained back-and-forth oscillation
        _flipWindowTicks++;
        if (_flipWindowTicks >= 40)
        {
            _flipWindowTicks = 0;
            _signFlips = Mathf.Max(0, _signFlips - 2);
        }

        bool isOscillating = _signFlips >= 8;
        bool isStalled = _ticksSinceProgress > 240; // ~12s without new best goal distance

        if (_mode != NavMode.EscapeTurn && (isOscillating || isStalled))
        {
            _signFlips = 0;
            _flipWindowTicks = 0;
            _ticksSinceProgress = 90;
            _escapeTicks = 28; // ~1.4s committed escape turn

            if (_mode == NavMode.FollowLeftWall)
            {
                _escapeDir = -1.0f;
            }
            else if (_mode == NavMode.FollowRightWall)
            {
                _escapeDir = 1.0f;
            }
            else
            {
                float leftRoom = l45 + 0.8f * l90;
                float rightRoom = r45 + 0.8f * r90;
                if (Mathf.Abs(leftRoom - rightRoom) < 0.10f)
                {
                    _escapeDir = obs.goalAngleDeg >= 0f ? 1.0f : -1.0f;
                }
                else
                {
                    _escapeDir = leftRoom >= rightRoom ? 1.0f : -1.0f;
                }
            }

            _mode = NavMode.EscapeTurn;
            _modeTicks = 0;
        }

        if (_mode == NavMode.EscapeTurn)
        {
            _escapeTicks--;
            float turnEsc = _escapeDir * 0.75f;
            float speedEsc = (f > 0.28f && l45 > 0.22f && r45 > 0.22f) ? 0.18f : 0.05f;

            if (_escapeTicks <= 0 || (f > 0.65f && l45 > 0.45f && r45 > 0.45f && _modeTicks >= 10))
            {
                if (_wallSwitchCount < 3 && (l90 < 0.65f || r90 < 0.65f))
                {
                    _wallSwitchCount++;
                    _mode = (_escapeDir > 0f) ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
                }
                else
                {
                    _mode = NavMode.SeekGoal;
                }
                _modeTicks = 0;
            }

            RecordTurnSign(turnEsc);
            return (speedEsc, turnEsc);
        }

        // Normal mode transitions
        bool frontObstacle = f < 0.42f || l45 < 0.30f || r45 < 0.30f;

        if (_mode == NavMode.SeekGoal)
        {
            if (frontObstacle && _modeTicks >= 3)
            {
                _mode = ChooseWallFollowMode(obs.goalAngleDeg, l45, r45, l90, r90);
                _modeTicks = 0;
            }
        }
        else // FollowLeftWall or FollowRightWall
        {
            bool leftWall = (_mode == NavMode.FollowLeftWall);
            float sideDist = leftWall ? l90 : r90;
            float diagWallDist = leftWall ? l45 : r45;
            float oppDiagDist = leftWall ? r45 : l45;

            if (_modeTicks >= 18)
            {
                bool lostWall = sideDist > 1.45f && diagWallDist > 1.35f && f > 0.85f;

                // Rejoin goal only when the direction toward the goal is clearly open
                bool goalTowardLeft = obs.goalAngleDeg > 8f;
                bool goalTowardRight = obs.goalAngleDeg < -8f;
                bool goalDirOpen =
                    (goalTowardLeft && f > 0.85f && l45 > 0.75f && oppDiagDist > 0.35f) ||
                    (goalTowardRight && f > 0.85f && r45 > 0.75f && oppDiagDist > 0.35f) ||
                    (!goalTowardLeft && !goalTowardRight && f > 0.90f && l45 > 0.55f && r45 > 0.55f);

                if (lostWall || (goalDirOpen && Mathf.Abs(obs.goalAngleDeg) < 55f))
                {
                    _mode = NavMode.SeekGoal;
                    _modeTicks = 0;
                }
            }
        }

        float speed01;
        float turn;

        if (_mode == NavMode.FollowLeftWall)
        {
            ComputeWallFollow(
                followLeft: true,
                f: f,
                diagSame: l45,
                diagOpp: r45,
                sideSame: l90,
                sideOpp: r90,
                out speed01,
                out turn
            );
        }
        else if (_mode == NavMode.FollowRightWall)
        {
            ComputeWallFollow(
                followLeft: false,
                f: f,
                diagSame: r45,
                diagOpp: l45,
                sideSame: r90,
                sideOpp: l90,
                out speed01,
                out turn
            );
        }
        else
        {
            ComputeSeekGoal(obs.goalAngleDeg, f, l45, r45, l90, r90, out speed01, out turn);
        }

        RecordTurnSign(turn);
        return (speed01, turn);
    }

    private NavMode ChooseWallFollowMode(float goalAngleDeg, float l45, float r45, float l90, float r90)
    {
        // If one side has a clear wall and the other is wide open, follow the side with more space to turn around
        if (l90 < 0.35f && r90 > 0.90f) return NavMode.FollowLeftWall;
        if (r90 < 0.35f && l90 > 0.90f) return NavMode.FollowRightWall;

        // Otherwise follow the wall on the side that lets us turn toward the goal without cutting across the obstacle
        if (goalAngleDeg < -10f)
        {
            // Goal is to the right: following left wall often guides around obstacle toward right,
            // or following right wall if right side is closer
            return (r45 < l45) ? NavMode.FollowRightWall : NavMode.FollowLeftWall;
        }
        if (goalAngleDeg > 10f)
        {
            return (l45 < r45) ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
        }

        return (l45 + l90) <= (r45 + r90) ? NavMode.FollowLeftWall : NavMode.FollowRightWall;
    }

    private void ComputeSeekGoal(
        float goalAngleDeg,
        float f,
        float l45,
        float r45,
        float l90,
        float r90,
        out float speed01,
        out float turn)
    {
        float goalTurn = Mathf.Clamp(goalAngleDeg / 36.0f, -1.0f, 1.0f);

        float leftRepulse = Repulsion(l45, 0.55f) * 1.1f + Repulsion(l90, 0.32f) * 0.6f;
        float rightRepulse = Repulsion(r45, 0.55f) * 1.1f + Repulsion(r90, 0.32f) * 0.6f;
        float frontRepulse = Repulsion(f, 0.68f);

        float avoidTurn = rightRepulse - leftRepulse;
        if (frontRepulse > 0.05f)
        {
            float turnSign;
            if (Mathf.Abs(rightRepulse - leftRepulse) > 0.12f)
            {
                turnSign = (rightRepulse > leftRepulse) ? 1.0f : -1.0f;
            }
            else
            {
                turnSign = (goalAngleDeg >= 0f) ? 1.0f : -1.0f;
            }
            avoidTurn += turnSign * frontRepulse * 1.1f;
        }

        float blend = Mathf.Clamp01(frontRepulse * 0.75f + Mathf.Max(leftRepulse, rightRepulse) * 0.45f);
        turn = Mathf.Clamp(goalTurn * (1.0f - blend) + avoidTurn, -1.0f, 1.0f);

        float minFront = Mathf.Min(f, Mathf.Min(l45, r45));
        float distScale = Mathf.Clamp01((minFront - 0.16f) / 0.60f);
        float turnScale = 1.0f - 0.45f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.90f * distScale * turnScale, 0.15f, 0.95f);
    }

    private void ComputeWallFollow(
        bool followLeft,
        float f,
        float diagSame,
        float diagOpp,
        float sideSame,
        float sideOpp,
        out float speed01,
        out float turn)
    {
        // +1 = left turn, -1 = right turn
        float s = followLeft ? 1.0f : -1.0f;
        const float desiredDist = 0.24f;

        // Obstacle ahead or corner on followed diagonal: turn away from followed wall
        if (f < 0.32f || diagSame < 0.25f)
        {
            turn = -s * 0.85f;
            speed01 = (f < 0.18f || diagSame < 0.16f) ? 0.08f : 0.25f;
            return;
        }

        // Narrow corridor: keep roughly centered between left and right walls
        if (sideSame < 0.45f && sideOpp < 0.45f)
        {
            float diff = sideOpp - sideSame; // > 0 means more room on opposite side
            turn = Mathf.Clamp(diff * 2.0f, -0.55f, 0.55f);
            speed01 = Mathf.Clamp(0.55f * Mathf.Clamp01((f - 0.22f) / 0.55f), 0.18f, 0.65f);
            return;
        }

        float err = sideSame - desiredDist;
        if (Mathf.Abs(err) < 0.02f) err = 0f;

        // Positive err (too far from wall) -> steer toward wall (+s); negative err -> steer away (-s)
        float wallCmd = s * Mathf.Clamp(err * 2.1f, -0.45f, 0.45f);

        if (diagSame < 0.42f)
        {
            wallCmd -= s * Mathf.Clamp((0.42f - diagSame) * 2.2f, 0f, 0.65f);
        }

        turn = Mathf.Clamp(wallCmd, -1.0f, 1.0f);

        float minClear = Mathf.Min(f, Mathf.Min(diagSame, Mathf.Min(diagOpp, sideSame)));
        float clearScale = Mathf.Clamp01((minClear - 0.16f) / 0.50f);
        float turnScale = 1.0f - 0.40f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.78f * clearScale * turnScale, 0.18f, 0.85f);
    }

    private static float Repulsion(float dist, float radius)
    {
        if (dist >= radius) return 0f;
        float d = Mathf.Max(dist, 0.02f);
        float x = (radius - d) / radius;
        return x * x;
    }

    private void RecordTurnSign(float turn)
    {
        if (Mathf.Abs(turn) > 0.18f && Mathf.Abs(_prevRawTurn) > 0.18f)
        {
            if (Mathf.Sign(turn) != Mathf.Sign(_prevRawTurn))
            {
                _signFlips++;
            }
        }
        _prevRawTurn = turn;
    }

    private float FilterSensor(int sensorIndex, float raw)
    {
        float val = Mathf.Clamp(raw, 0.02f, MaxSensorRange);

        // Guard against occasional false 4.0m readings when recent readings showed a close object
        if (val >= MaxSensorRange - 1e-4f && _histCount[sensorIndex] > 0)
        {
            float avg = ComputeAverage(sensorIndex);
            if (avg < 1.25f)
            {
                val = Mathf.Clamp(avg + 0.12f, 0.02f, 1.6f);
            }
        }

        int idx = _histIdx[sensorIndex];
        _hist[sensorIndex, idx] = val;
        _histIdx[sensorIndex] = (idx + 1) % HistorySize;
        if (_histCount[sensorIndex] < HistorySize)
        {
            _histCount[sensorIndex]++;
        }

        int count = _histCount[sensorIndex];
        float[] sorted = new float[count];
        for (int i = 0; i < count; i++)
        {
            sorted[i] = _hist[sensorIndex, i];
        }
        Array.Sort(sorted);
        float median = sorted[count / 2];

        float mean = ComputeAverage(sensorIndex);
        return Mathf.Clamp(0.65f * median + 0.35f * mean, 0.02f, MaxSensorRange);
    }

    private float ComputeAverage(int sensorIndex)
    {
        int count = _histCount[sensorIndex];
        if (count == 0) return MaxSensorRange;
        float sum = 0f;
        for (int i = 0; i < count; i++)
        {
            sum += _hist[sensorIndex, i];
        }
        return sum / count;
    }

Test report: 6/10 reached the goal, 0 collisions, 4 timeouts (got stuck), 0 runtime errors. Most common failure: gets stuck oscillating.

Oscillation in a reactive controller without odometry usually happens when two competing behaviors fight at a boundary:

Premature exit from wall-following back to SeekGoal: If the rover exits wall-following as soon as goalAngleDeg looks reasonable, it steers back into the same obstacle or U-shape pocket, triggers wall-following again, and loops.

Over-sensitive steering near walls/corridors: High gains on small distance or angle errors make the rover swing past its target line and correct back repeatedly—especially with up to 10% Gaussian sensor noise.

Late or short unstick maneuvers: Brief escape turns that immediately hand control back to SeekGoal often land right back in the same trap.

Key Changes for Stability
Stronger wall-follow commitment: Once following a left or right wall, stay on that wall until the goal is roughly ahead (|goalAngleDeg| < 35°) and front + diagonal sensors are wide open (> 1.0 m front, > 0.75 m diagonals), reducing SeekGoal ↔ WallFollow chatter.

Calmer, deadbanded steering: Wider angle/offset divisors and small deadbands so minor sensor noise doesn't trigger constant left/right corrections.

Longer, cleaner stall/oscillation recovery: When progress stalls or turn reversals spike, commit to a steady turn/wall-follow switch for ~2.5 seconds (50 ticks) with a cooldown before re-triggering.

        _unstickTicks = 0;
        _unstickCooldown = 0;
        _unstickTurn = 0.7f;

        _bestGoalDist = float.MaxValue;
        _ticksNoProgress = 0;

        _lastTurn = 0f;
        _reversalCount = 0;
        _reversalDecayTick = 0;

            _bufPos[i] = 0;

            }
        }
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        _modeTicks++;
        if (_unstickCooldown > 0) _unstickCooldown--;

        if (obs.goalDistance < _bestGoalDist - 0.05f)
        {
            _bestGoalDist = obs.goalDistance;
            _ticksNoProgress = 0;
        }
        else
        {
            _ticksNoProgress++;
        }

        float f = ReadFiltered(0, obs.front);
        float l45 = ReadFiltered(1, obs.left);
        float r45 = ReadFiltered(2, obs.right);
        float l90 = ReadFiltered(3, obs.leftSide);
        float r90 = ReadFiltered(4, obs.rightSide);

        float minFront = Mathf.Min(f, Mathf.Min(l45, r45));

        // Goal close and unobstructed
        if (obs.goalDistance < 0.15f && minFront > 0.24f)
        {
            float t = Mathf.Abs(obs.goalAngleDeg) < 4f ? 0f : Mathf.Clamp(obs.goalAngleDeg / 45f, -0.8f, 0.8f);
            float s = Mathf.Clamp01(0.40f * (1f - 0.4f * Mathf.Abs(t)));
            TrackReversals(t);
            return (s, t);
        }

        // Decay reversal counter gradually
        _reversalDecayTick++;
        if (_reversalDecayTick >= 25)
        {
            _reversalDecayTick = 0;
            _reversalCount = Mathf.Max(0, _reversalCount - 2);
        }

        // Trigger Unstick only when genuinely stuck or oscillating, respecting cooldown
        bool stuckByTime = _ticksNoProgress > 220; // ~11s without new best goal distance
        bool stuckByOsc = _reversalCount >= 10;

        if (_mode != Mode.Unstick && _unstickCooldown == 0 && (stuckByTime || stuckByOsc))
        {
            _reversalCount = 0;
            _ticksNoProgress = 80;
            _unstickCooldown = 80; // ~4s cooldown
            _unstickTicks = 50;    // ~2.5s committed maneuver

            if (_mode == Mode.FollowLeft)
            {
                _unstickTurn = -0.65f;
            }
            else if (_mode == Mode.FollowRight)
            {
                _unstickTurn = 0.65f;
            }
            else
            {
                float leftOpen = l45 + 0.75f * l90;
                float rightOpen = r45 + 0.75f * r90;
                if (Mathf.Abs(leftOpen - rightOpen) < 0.12f)
                {
                    _unstickTurn = obs.goalAngleDeg >= 0f ? 0.65f : -0.65f;
                }
                else
                {
                    _unstickTurn = leftOpen > rightOpen ? 0.65f : -0.65f;
                }
            }

            _mode = Mode.Unstick;
            _modeTicks = 0;
        }

        if (_mode == Mode.Unstick)
        {
            _unstickTicks--;
            float turnEsc = _unstickTurn;
            if (f < 0.22f)
            {
                // Turn harder in place if very close to an obstacle
                turnEsc = Mathf.Sign(_unstickTurn) * 0.85f;
            }
            float speedEsc = (f > 0.30f && l45 > 0.24f && r45 > 0.24f) ? 0.25f : 0.06f;

            if (_unstickTicks <= 0)
            {
                if (l90 < 0.55f && _unstickTurn > 0f)
                {
                    _mode = Mode.FollowLeft;
                }
                else if (r90 < 0.55f && _unstickTurn < 0f)
                {
                    _mode = Mode.FollowRight;
                }
                else
                {
                    _mode = Mode.SeekGoal;
                }
                _modeTicks = 0;
            }

            TrackReversals(turnEsc);
            return (speedEsc, turnEsc);
        }

        // Mode transitions with strong hysteresis
        bool blockedAhead = f < 0.44f || l45 < 0.32f || r45 < 0.32f;

        if (_mode == Mode.SeekGoal)
        {
            if (blockedAhead && _modeTicks >= 5)
            {
                _mode = PickWallMode(obs.goalAngleDeg, l45, r45, l90, r90);
                _modeTicks = 0;
            }
        }
        else // FollowLeft or FollowRight
        {
            bool isLeft = (_mode == Mode.FollowLeft);
            float sideDist = isLeft ? l90 : r90;
            float diagWall = isLeft ? l45 : r45;

            if (_modeTicks >= 24) // Stay on wall at least ~1.2s
            {
                bool wallGone = sideDist > 1.50f && diagWall > 1.35f && f > 0.90f;

                bool goalRoughlyAhead = Mathf.Abs(obs.goalAngleDeg) < 35f;
                bool goalSideClear = obs.goalAngleDeg >= 0f ? l45 > 0.78f : r45 > 0.78f;
                bool clearToRejoin = goalRoughlyAhead && goalSideClear && f > 1.00f && l45 > 0.55f && r45 > 0.55f;

                if (wallGone || clearToRejoin)
                {
                    _mode = Mode.SeekGoal;
                    _modeTicks = 0;
                }
            }
        }

        float speed01;
        float turn;

        if (_mode == Mode.FollowLeft)
        {
            SteerWall(followLeft: true, f, l45, r45, l90, r90, out speed01, out turn);
        }
        else if (_mode == Mode.FollowRight)
        {
            SteerWall(followLeft: false, f, r45, l45, r90, l90, out speed01, out turn);
        }
        else
        {
            SteerSeek(obs.goalAngleDeg, f, l45, r45, l90, r90, out speed01, out turn);
        }

        TrackReversals(turn);
        return (speed01, turn);
    }

    private static Mode PickWallMode(float goalAngleDeg, float l45, float r45, float l90, float r90)
    {
        if (l90 < 0.35f && r90 > 0.85f) return Mode.FollowLeft;
        if (r90 < 0.35f && l90 > 0.85f) return Mode.FollowRight;

        if (goalAngleDeg < -15f)
        {
            return r45 < l45 - 0.10f ? Mode.FollowRight : Mode.FollowLeft;
        }
        if (goalAngleDeg > 15f)
        {
            return l45 < r45 - 0.10f ? Mode.FollowLeft : Mode.FollowRight;
        }

        return (l45 + l90) < (r45 + r90) ? Mode.FollowLeft : Mode.FollowRight;
    }

    private void SteerSeek(
        float goalAngleDeg,
        float f,
        float l45,
        float r45,
        float l90,
        float r90,
        out float speed01,
        out float turn)
    {
        float angleCmd = Mathf.Abs(goalAngleDeg) < 3.5f
            ? 0f
            : Mathf.Clamp(goalAngleDeg / 48f, -0.85f, 0.85f);

        float leftPush = SoftPush(l45, 0.55f) * 0.9f + SoftPush(l90, 0.30f) * 0.5f;
        float rightPush = SoftPush(r45, 0.55f) * 0.9f + SoftPush(r90, 0.30f) * 0.5f;
        float frontPush = SoftPush(f, 0.70f);

        float avoidCmd = Mathf.Clamp(rightPush - leftPush, -0.65f, 0.65f);
        if (frontPush > 0.05f)
        {
            float dir = Mathf.Abs(rightPush - leftPush) > 0.15f
                ? (rightPush > leftPush ? 1f : -1f)
                : (goalAngleDeg >= 0f ? 1f : -1f);
            avoidCmd += dir * frontPush * 0.85f;
        }

        float mix = Mathf.Clamp01(frontPush * 0.8f + Mathf.Max(leftPush, rightPush) * 0.4f);
        turn = Mathf.Clamp(angleCmd * (1f - mix) + avoidCmd, -1f, 1f);

        float minFront = Mathf.Min(f, Mathf.Min(l45, r45));
        float distFactor = Mathf.Clamp01((minFront - 0.18f) / 0.55f);
        float turnFactor = 1f - 0.45f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.80f * distFactor * turnFactor, 0.16f, 0.90f);
    }

    private void SteerWall(
        bool followLeft,
        float f,
        float diagSame,
        float diagOpp,
        float sideSame,
        float sideOpp,
        out float speed01,
        out float turn)
    {
        // +1 = turn left, -1 = turn right
        float s = followLeft ? 1f : -1f;
        const float targetDist = 0.25f;

        if (f < 0.30f || diagSame < 0.23f)
        {
            turn = -s * 0.75f;
            speed01 = (f < 0.18f || diagSame < 0.15f) ? 0.08f : 0.22f;
            return;
        }

        // Corridor centering with deadband
        if (sideSame < 0.44f && sideOpp < 0.44f)
        {
            float diff = sideOpp - sideSame;
            turn = Mathf.Abs(diff) < 0.04f ? 0f : Mathf.Clamp(diff * 1.4f, -0.40f, 0.40f);
            speed01 = Mathf.Clamp(0.48f * Mathf.Clamp01((f - 0.24f) / 0.50f), 0.18f, 0.55f);
            return;
        }

        float err = sideSame - targetDist;
        if (Mathf.Abs(err) < 0.03f) err = 0f;

        float cmd = s * Mathf.Clamp(err * 1.5f, -0.35f, 0.35f);

        if (diagSame < 0.40f)
        {
            cmd -= s * Mathf.Clamp((0.40f - diagSame) * 1.6f, 0f, 0.50f);
        }

        turn = Mathf.Clamp(cmd, -1f, 1f);

        float minClear = Mathf.Min(f, Mathf.Min(diagSame, Mathf.Min(diagOpp, sideSame)));
        float distFactor = Mathf.Clamp01((minClear - 0.18f) / 0.50f);
        float turnFactor = 1f - 0.35f * Mathf.Abs(turn);
        speed01 = Mathf.Clamp(0.70f * distFactor * turnFactor, 0.18f, 0.75f);
    }

    private static float SoftPush(float dist, float radius)
    {
        if (dist >= radius) return 0f;
        float d = Mathf.Max(dist, 0.02f);
        float x = (radius - d) / radius;
        return x * x;
    }

    private void TrackReversals(float turn)
    {
        if (Mathf.Abs(turn) > 0.22f && Mathf.Abs(_lastTurn) > 0.22f)
        {
            if (Mathf.Sign(turn) != Mathf.Sign(_lastTurn))
            {
                _reversalCount++;
            }
        }
        _lastTurn = turn;
    }

    private float ReadFiltered(int sensorIdx, float raw)
    {
        float v = Mathf.Clamp(raw, 0.02f, MaxRange);

        // If a reading suddenly jumps to 4.0 while recent readings were close, treat as a likely false 4.0
        if (v >= MaxRange - 1e-4f && _bufLen[sensorIdx] > 0)
        {
            float sum = 0f;
            for (int i = 0; i < _bufLen[sensorIdx]; i++)
            {
                sum += _buf[sensorIdx, i];
            }
            float avg = sum / _bufLen[sensorIdx];
            if (avg < 1.0f)
            {
                v = avg;
            }
        }

        int pos = _bufPos[sensorIdx];
        _buf[sensorIdx, pos] = v;
        _bufPos[sensorIdx] = (pos + 1) % Window;
        if (_bufLen[sensorIdx] < Window) _bufLen[sensorIdx]++;

        int len = _bufLen[sensorIdx];
        float[] copy = new float[len];
        for (int i = 0; i < len; i++)
        {
            copy[i] = _buf[sensorIdx, i];
        }
        Array.Sort(copy);
        return copy[len / 2];
    }
