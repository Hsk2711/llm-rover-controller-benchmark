You are writing a navigation controller for a small simulated ground rover in Unity (C#).

TASK: Drive the rover from its start position to a goal area without hitting any obstacle
or wall. The environment is a 6 m x 6 m arena with unknown box-shaped obstacles, which
may include narrow corridors and U-shaped dead ends. You have 60 seconds.
Any contact with an obstacle or wall ends the run as a failure.

SENSORS (all you get, updated 20 times per second):
- Five ultrasonic distance sensors, readings in metres:
  front (0 deg), left (+45 deg), right (-45 deg), leftSide (+90 deg), rightSide (-90 deg),
  angles measured from the rover's heading, positive = to the left.
  The three forward sensors sit on the front face; the two side sensors sit at the middle
  of each side. Each has a cone of about +-15 deg and reports the nearest object in it.
  Range 0.02 to 4.0 m; 4.0 means nothing detected. Readings are NOISY (Gaussian noise
  proportional to distance, up to 10%) and occasionally a reading falsely returns 4.0.
- goalDistance: straight-line distance to the goal centre in metres.
- goalAngleDeg: signed angle from the rover's heading to the goal, -180..180
  (positive = goal is to the left).
There is no map, no position, no camera and no lidar.

ACTUATION: return (speed01, turn) every 0.05 s.
- speed01: 0..1, fraction of max forward speed 0.3 m/s (no reversing).
- turn: -1..1, fraction of max turn rate 90 deg/s (positive = turn left).
Rover size: 0.20 m long x 0.15 m wide.

WRITE exactly one C# class, named LLM_Controller, that implements the interface below.
These types ALREADY EXIST in the project (global namespace) - do NOT declare them again.
Observation has these public float fields for you to use (it may contain other fields;
ignore them): front, left, right, leftSide, rightSide, goalDistance, goalAngleDeg.

public interface IRoverController {
    string Name { get; }
    void ResetController();   // called at the start of every run
    (float speed01, float turn) Decide(Observation obs);   // called every 0.05 s
}

RULES: Plain C# only (System and UnityEngine.Mathf allowed). Do not use
UnityEngine.Random, Physics, GameObject, Transform, Time or any Unity scene API.
Keep any memory you need in private fields and clear it in ResetController().
The controller must be deterministic. Output the complete code in one block.

Below is a reactive controller based on the DistBug algorithm. It has two modes:

Seek heads straight for the goal and steers away from nearby walls.
Follow tracks an obstacle's boundary with a PD controller once the way ahead is blocked.

It switches back to Seek when that is provably progress. Sensor noise and false 4.0 m readings are removed with a 5-sample median filter. Distance thresholds are set relative to the rover's footprint, so it is safe to turn in place.

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "LLM_Controller";

    // ---------------- constants ----------------
    const float DT = 0.05f;
    const int FILT = 5;                 // median filter length

    const float EMERG_FRONT = 0.12f;    // stop & rotate thresholds
    const float EMERG_DIAG = 0.08f;
    const float EMERG_SIDE = 0.035f;
    const float ESCAPE_CLEAR_FRONT = 0.30f;
    const float ESCAPE_CLEAR_DIAG = 0.18f;

    const float BLOCK_FRONT = 0.45f;    // obstacle ahead -> wall-follow
    const float WALL_TARGET = 0.22f;    // desired lateral distance to wall
    const float WALL_LOST = 0.70f;      // wall considered lost beyond this
    const float LEAVE_STEP = 0.50f;     // DistBug progress margin
    const float FOLLOW_TIMEOUT = 22f;   // switch follow side after this
    const float GOAL_STOP = 0.06f;

    // sensor indices
    const int S_F = 0, S_L = 1, S_R = 2, S_LS = 3, S_RS = 4;

    // ---------------- state ----------------
    float[,] buf = new float[5, FILT];
    int bufIdx;
    int samples;
    float[] tmp = new float[FILT];

    enum Mode { Seek, Follow }
    Mode mode;
    int followSide;          // +1 = wall on left, -1 = wall on right
    float hitDist;
    float followTime;
    int escapeDir;           // 0 = not escaping, +1 left, -1 right
    float prevWall;
    bool havePrevWall;
    int sideSwitches;

    public LLM_Controller()
    {
        ResetController();
    }

    public void ResetController()
    {
        buf = new float[5, FILT];
        tmp = new float[FILT];
        bufIdx = 0;
        samples = 0;
        mode = Mode.Seek;
        followSide = -1;
        hitDist = float.MaxValue;
        followTime = 0f;
        escapeDir = 0;
        prevWall = 0f;
        havePrevWall = false;
        sideSwitches = 0;
    }

    // ---------------- filtering ----------------
    void Push(Observation o)
    {
        float[] v = { o.front, o.left, o.right, o.leftSide, o.rightSide };
        if (samples == 0)
        {
            for (int s = 0; s < 5; s++)
                for (int k = 0; k < FILT; k++) buf[s, k] = Sanitize(v[s]);
        }
        for (int s = 0; s < 5; s++) buf[s, bufIdx] = Sanitize(v[s]);
        bufIdx = (bufIdx + 1) % FILT;
        if (samples < FILT) samples++;
    }

    static float Sanitize(float x)
    {
        if (float.IsNaN(x) || float.IsInfinity(x)) return 4f;
        return Mathf.Clamp(x, 0.02f, 4f);
    }

    float Med(int s)
    {
        for (int k = 0; k < FILT; k++) tmp[k] = buf[s, k];
        Array.Sort(tmp);
        return tmp[FILT / 2];
    }

    // ---------------- main ----------------
    public (float speed01, float turn) Decide(Observation obs)
    {
        Push(obs);
        float f = Med(S_F), l = Med(S_L), r = Med(S_R), ls = Med(S_LS), rs = Med(S_RS);
        float gd = obs.goalDistance;
        float ga = obs.goalAngleDeg;

        if (gd < GOAL_STOP) return (0f, 0f);

        // ---------- emergency: stop and rotate in place ----------
        bool emergency = f < EMERG_FRONT || l < EMERG_DIAG || r < EMERG_DIAG;
        if (emergency && escapeDir == 0)
        {
            if (mode == Mode.Follow)
                escapeDir = -followSide;                 // rotate away from followed wall
            else
                escapeDir = (l + 0.5f * ls >= r + 0.5f * rs) ? 1 : -1;

            if (mode == Mode.Seek) EnterFollow(-escapeDir, gd);
        }
        if (escapeDir != 0)
        {
            if (f > ESCAPE_CLEAR_FRONT && l > ESCAPE_CLEAR_DIAG && r > ESCAPE_CLEAR_DIAG)
                escapeDir = 0;
            else
            {
                if (mode == Mode.Follow) followTime += DT;
                return (0f, 0.8f * escapeDir);
            }
        }

        if (mode == Mode.Follow)
        {
            followTime += DT;

            if (CanLeave(f, l, r, ls, rs, gd, ga))
            {
                mode = Mode.Seek;
                havePrevWall = false;
            }
            else
            {
                if (followTime > FOLLOW_TIMEOUT)
                {
                    // probably circling; try the other way round
                    followSide = -followSide;
                    followTime = 0f;
                    hitDist = gd;
                    havePrevWall = false;
                    sideSwitches++;
                }
                return FollowWall(f, l, r, ls, rs, gd);
            }
        }

        return Seek(f, l, r, ls, rs, gd, ga);
    }

    // ---------------- goal seeking ----------------
    (float, float) Seek(float f, float l, float r, float ls, float rs, float gd, float ga)
    {
        // blocked ahead -> start following the obstacle
        if (f < BLOCK_FRONT && Mathf.Abs(ga) < 80f)
        {
            int dir;   // +1 = go around by turning left
            bool goalLeft = ga >= 0f;
            float leftSpace = l + 0.5f * ls;
            float rightSpace = r + 0.5f * rs;
            if (goalLeft) dir = (leftSpace < 0.35f && rightSpace > leftSpace + 0.3f) ? -1 : 1;
            else dir = (rightSpace < 0.35f && leftSpace > rightSpace + 0.3f) ? 1 : -1;

            EnterFollow(-dir, gd);
            return FollowWall(f, l, r, ls, rs, gd);
        }

        // goal far behind: rotate (almost) in place
        if (Mathf.Abs(ga) > 70f)
        {
            float sp = (f > 0.6f && l > 0.3f && r > 0.3f) ? 0.1f : 0f;
            return (sp, Mathf.Sign(ga) * 0.9f);
        }

        float turn = Mathf.Clamp(ga / 40f, -1f, 1f);

        // repulsion from close obstacles (keeps centered in corridors)
        float rep = 0f;
        if (l < 0.5f) rep -= (0.5f - l) * 2.0f;
        if (r < 0.5f) rep += (0.5f - r) * 2.0f;
        if (ls < 0.2f) rep -= (0.2f - ls) * 2.5f;
        if (rs < 0.2f) rep += (0.2f - rs) * 2.5f;
        turn = Mathf.Clamp(turn + rep, -1f, 1f);

        float minFwd = Mathf.Min(f, Mathf.Min(l, r) * 1.2f);
        float speed = Mathf.Clamp((minFwd - EMERG_FRONT) / 0.6f, 0.15f, 1f);
        speed *= 1f - 0.6f * Mathf.Abs(turn);
        speed = Mathf.Min(speed, 0.25f + gd / 0.4f);
        return (Mathf.Clamp01(speed), turn);
    }

    void EnterFollow(int side, float gd)
    {
        if (mode != Mode.Follow)
        {
            mode = Mode.Follow;
            followSide = side;
            hitDist = gd;
            followTime = 0f;
            havePrevWall = false;
        }
    }

    // ---------------- wall following ----------------
    (float, float) FollowWall(float f, float l, float r, float ls, float rs, float gd)
    {
        int s = followSide;
        float wallSide = s > 0 ? ls : rs;
        float wallDiag = s > 0 ? l : r;
        float otherSide = s > 0 ? rs : ls;
        float otherDiag = s > 0 ? r : l;

        // perpendicular-ish distance estimate to the followed wall
        float dWall = Mathf.Min(wallSide, wallDiag * 0.72f);

        // narrow corridor: aim for the centre instead of a fixed offset
        float target = WALL_TARGET;
        if (otherSide < 0.5f) target = Mathf.Min(WALL_TARGET, 0.5f * (wallSide + otherSide));

        float turn, speed;

        if (f < BLOCK_FRONT)
        {
            // inner corner: turn away from the wall, creep forward only if room
            turn = -s * 1.0f;
            speed = Mathf.Clamp((f - EMERG_FRONT - 0.08f) / 0.6f, 0f, 0.25f);
            havePrevWall = false;
        }
        else if (dWall > WALL_LOST)
        {
            // outer corner / wall lost: arc around towards the wall side
            turn = s * 0.7f;
            speed = 0.45f;
            havePrevWall = false;
        }
        else
        {
            float err = dWall - target;
            float deriv = 0f;
            if (havePrevWall) deriv = Mathf.Clamp((dWall - prevWall) / DT, -0.5f, 0.5f);
            prevWall = dWall;
            havePrevWall = true;

            turn = s * (3.0f * err + 1.2f * deriv);

            // push away from things on the opposite side as well
            if (otherDiag < 0.25f) turn += s * (0.25f - otherDiag) * 3f;
            if (otherSide < 0.12f) turn += s * (0.12f - otherSide) * 4f;

            turn = Mathf.Clamp(turn, -0.85f, 0.85f);
            speed = Mathf.Clamp((f - EMERG_FRONT) / 0.5f, 0.2f, 0.9f);
            speed *= 1f - 0.5f * Mathf.Abs(turn);
        }

        // never drive fast with a diagonal very close
        float minDiag = Mathf.Min(l, r);
        if (minDiag < 0.2f) speed = Mathf.Min(speed, 0.25f);

        return (Mathf.Clamp01(speed), Mathf.Clamp(turn, -1f, 1f));
    }

    // ---------------- DistBug leave condition ----------------
    bool CanLeave(float f, float l, float r, float ls, float rs, float gd, float ga)
    {
        if (followTime < 1.0f) return false;
        float a = Mathf.Abs(ga);
        if (a > 100f) return false;

        float free;
        if (a <= 22.5f) free = f;
        else if (a <= 67.5f) free = ga > 0f ? Mathf.Min(l, f + 0.2f) : Mathf.Min(r, f + 0.2f);
        else free = ga > 0f ? Mathf.Min(ls, l + 0.2f) : Mathf.Min(rs, r + 0.2f);

        // goal side must not be the wall we are hugging at close range
        bool goalTowardWall = (followSide > 0 && ga > 30f) || (followSide < 0 && ga < -30f);
        if (goalTowardWall && free < 0.6f) return false;

        free *= 0.9f; // account for up to 10% noise

        if (free < BLOCK_FRONT + 0.15f && free < gd) return false;

        // goal visible with clear line, or guaranteed progress (DistBug)
        if (free >= gd) return true;
        if (gd - free <= hitDist - LEAVE_STEP) return true;
        return false;
    }
}

How it works:

Filtering: each sensor uses the median of its last 5 readings. This drops the occasional false 4.0 readings and smooths the noise.
Safety: if the front reading is under 0.12 m or a diagonal is under 0.08 m, the rover stops and rotates in place. It keeps rotating the same way, away from the followed wall, until there is clear space. Rotating in place is safe because the rover's corners sit only 0.125 m from its centre. It never reverses.
Seek mode: steers proportionally toward goalAngleDeg and is pushed away by nearby diagonal and side readings, which keeps it centred in narrow corridors. Speed scales with free space ahead.
Follow mode: keeps the wall at about 0.22 m with PD control, or at the centre in tight corridors. It turns away at inner corners and arcs around outer corners.
Leaving an obstacle: the rover returns to Seek when the path to the goal is clear, or when the DistBug rule guarantees progress. That rule requires the goal distance minus the free space toward the goal to be at least 0.5 m below the distance recorded at the hit point. This is what gets it out of U-shaped dead ends without looping.
Timeout: if it follows one boundary for more than 22 s, it switches to following the other side as a fallback. Test report: 8/10 reached the goal, 2 collisions, 0 timeouts (got stuck), 0 runtime errors. Most common failure: hits obstacles head-on.

I made five changes to stop the head-on collisions:

Faster reaction: the 5-sample median added about 0.1 s of delay before the rover noticed an obstacle coming closer. Safety decisions now use a fast conservative reading instead. It is the smallest of the last 3 samples, ignoring false 4.0 readings.
Speed tied to stopping distance: forward speed now drops toward zero as the front or diagonal clearance shrinks. Previously Seek mode kept a minimum speed of 0.15 even when an obstacle was very close.
Blind gaps between sensor cones: the cones leave gaps at roughly 15–30° off the heading. When something is near a front corner, the rover now slows down and turns away more strongly.
Earlier, wider margins: emergency thresholds are larger. The rover switches to wall-following sooner, also when a diagonal sensor shows an obstacle on the goal side. Inner corners are taken almost in place.
Exiting a spin: after rotating in place, it only moves forward again once there is clearly room.

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "LLM_Controller";

    // ---------------- constants ----------------
    const float DT = 0.05f;
    const int FILT = 5;                  // median filter length (control)
    const int CONS = 3;                  // conservative window (safety)

    const float EMERG_FRONT = 0.16f;     // stop & rotate thresholds
    const float EMERG_DIAG = 0.10f;
    const float ESCAPE_CLEAR_FRONT = 0.38f;
    const float ESCAPE_CLEAR_DIAG = 0.22f;

    const float BLOCK_FRONT = 0.50f;     // obstacle ahead -> wall-follow
    const float BLOCK_DIAG = 0.26f;      // obstacle on goal-side diagonal
    const float WALL_TARGET = 0.24f;     // desired lateral distance to wall
    const float WALL_LOST = 0.70f;
    const float LEAVE_STEP = 0.50f;      // DistBug progress margin
    const float FOLLOW_TIMEOUT = 22f;
    const float GOAL_STOP = 0.06f;

    const int S_F = 0, S_L = 1, S_R = 2, S_LS = 3, S_RS = 4;

    // ---------------- state ----------------
    float[,] buf = new float[5, FILT];
    int bufIdx;
    int samples;
    float[] tmp = new float[FILT];

    enum Mode { Seek, Follow }
    Mode mode;
    int followSide;          // +1 = wall on left, -1 = wall on right
    float hitDist;
    float followTime;
    int escapeDir;           // 0 = not escaping, +1 left, -1 right
    float prevWall;
    bool havePrevWall;

    public LLM_Controller()
    {
        ResetController();
    }

    public void ResetController()
    {
        buf = new float[5, FILT];
        tmp = new float[FILT];
        bufIdx = 0;
        samples = 0;
        mode = Mode.Seek;
        followSide = -1;
        hitDist = float.MaxValue;
        followTime = 0f;
        escapeDir = 0;
        prevWall = 0f;
        havePrevWall = false;
    }

    // ---------------- filtering ----------------
    void Push(Observation o)
    {
        float[] v = { o.front, o.left, o.right, o.leftSide, o.rightSide };
        if (samples == 0)
        {
            for (int s = 0; s < 5; s++)
                for (int k = 0; k < FILT; k++) buf[s, k] = Sanitize(v[s]);
        }
        for (int s = 0; s < 5; s++) buf[s, bufIdx] = Sanitize(v[s]);
        bufIdx = (bufIdx + 1) % FILT;
        if (samples < FILT) samples++;
    }

    static float Sanitize(float x)
    {
        if (float.IsNaN(x) || float.IsInfinity(x)) return 4f;
        return Mathf.Clamp(x, 0.02f, 4f);
    }

    // median of last FILT samples (smooth, for steering)
    float Med(int s)
    {
        for (int k = 0; k < FILT; k++) tmp[k] = buf[s, k];
        Array.Sort(tmp);
        return tmp[FILT / 2];
    }

    // minimum of last CONS samples, ignoring false "nothing" readings (fast, for safety)
    float Cons(int s)
    {
        float m = 4f;
        bool any = false;
        for (int i = 1; i <= CONS; i++)
        {
            int idx = (bufIdx - i + FILT * 2) % FILT;
            float x = buf[s, idx];
            if (x < 3.95f) { any = true; if (x < m) m = x; }
        }
        return any ? m : 4f;
    }

    // ---------------- main ----------------
    public (float speed01, float turn) Decide(Observation obs)
    {
        Push(obs);

        // smoothed (steering)
        float f = Med(S_F), l = Med(S_L), r = Med(S_R), ls = Med(S_LS), rs = Med(S_RS);
        // conservative (safety)
        float fc = Cons(S_F), lc = Cons(S_L), rc = Cons(S_R), lsc = Cons(S_LS), rsc = Cons(S_RS);

        float gd = obs.goalDistance;
        float ga = obs.goalAngleDeg;

        if (gd < GOAL_STOP) return (0f, 0f);

        // ---------- emergency: stop and rotate in place ----------
        bool emergency = fc < EMERG_FRONT || lc < EMERG_DIAG || rc < EMERG_DIAG;
        if (emergency && escapeDir == 0)
        {
            if (mode == Mode.Follow)
                escapeDir = -followSide;
            else
                escapeDir = (lc + 0.5f * lsc >= rc + 0.5f * rsc) ? 1 : -1;

            if (mode == Mode.Seek) EnterFollow(-escapeDir, gd);
        }
        if (escapeDir != 0)
        {
            if (fc > ESCAPE_CLEAR_FRONT && lc > ESCAPE_CLEAR_DIAG && rc > ESCAPE_CLEAR_DIAG)
                escapeDir = 0;
            else
            {
                if (mode == Mode.Follow) followTime += DT;
                havePrevWall = false;
                return (0f, 0.8f * escapeDir);
            }
        }

        if (mode == Mode.Follow)
        {
            followTime += DT;

            if (CanLeave(fc, lc, rc, lsc, rsc, gd, ga))
            {
                mode = Mode.Seek;
                havePrevWall = false;
            }
            else
            {
                if (followTime > FOLLOW_TIMEOUT)
                {
                    followSide = -followSide;
                    followTime = 0f;
                    hitDist = gd;
                    havePrevWall = false;
                }
                return FollowWall(f, l, r, ls, rs, fc, lc, rc, lsc, rsc);
            }
        }

        return Seek(f, l, r, ls, rs, fc, lc, rc, lsc, rsc, gd, ga);
    }

    // speed limit from clearance (so the rover can always stop in time)
    static float SafeSpeed(float fc, float lc, float rc)
    {
        float sF = Mathf.Clamp01((fc - EMERG_FRONT) / 0.55f);
        float sD = Mathf.Clamp01((Mathf.Min(lc, rc) - EMERG_DIAG) / 0.40f);
        return Mathf.Min(sF, sD);
    }

    // ---------------- goal seeking ----------------
    (float, float) Seek(float f, float l, float r, float ls, float rs,
                        float fc, float lc, float rc, float lsc, float rsc,
                        float gd, float ga)
    {
        bool goalLeft = ga >= 0f;
        bool frontBlocked = fc < BLOCK_FRONT && Mathf.Abs(ga) < 90f;
        bool diagBlocked = (goalLeft && ga > 15f && ga < 90f && lc < BLOCK_DIAG) ||
                           (!goalLeft && ga < -15f && ga > -90f && rc < BLOCK_DIAG);
        // don't treat narrow-corridor diagonals as a block if straight ahead is open
        if (diagBlocked && fc > 0.8f && Mathf.Abs(ga) < 35f) diagBlocked = false;

        if (frontBlocked || diagBlocked)
        {
            int dir;   // +1 = go around by turning left
            float leftSpace = lc + 0.5f * lsc;
            float rightSpace = rc + 0.5f * rsc;
            if (frontBlocked)
            {
                if (goalLeft) dir = (leftSpace < 0.4f && rightSpace > leftSpace + 0.3f) ? -1 : 1;
                else dir = (rightSpace < 0.4f && leftSpace > rightSpace + 0.3f) ? 1 : -1;
            }
            else
            {
                // obstacle sits on goal-side diagonal: go around it the other way round
                dir = goalLeft ? -1 : 1;
                if (goalLeft && rightSpace < 0.3f) dir = 1;
                if (!goalLeft && leftSpace < 0.3f) dir = -1;
            }
            EnterFollow(-dir, gd);
            return FollowWall(f, l, r, ls, rs, fc, lc, rc, lsc, rsc);
        }

        // goal far to the side/behind: rotate in place
        if (Mathf.Abs(ga) > 60f)
            return (0f, Mathf.Sign(ga) * 0.9f);

        float turn = Mathf.Clamp(ga / 40f, -1f, 1f);

        // repulsion (uses conservative readings), keeps centred in corridors
        float rep = 0f;
        if (lc < 0.55f) rep -= (0.55f - lc) * 2.4f;
        if (rc < 0.55f) rep += (0.55f - rc) * 2.4f;
        if (lsc < 0.2f) rep -= (0.2f - lsc) * 3f;
        if (rsc < 0.2f) rep += (0.2f - rsc) * 3f;
        turn = Mathf.Clamp(turn + rep, -1f, 1f);

        float speed = SafeSpeed(fc, lc, rc);
        speed *= 1f - 0.6f * Mathf.Abs(turn);
        speed = Mathf.Min(speed, 0.2f + gd / 0.4f);
        return (Mathf.Clamp01(speed), turn);
    }

    void EnterFollow(int side, float gd)
    {
        if (mode != Mode.Follow)
        {
            mode = Mode.Follow;
            followSide = side;
            hitDist = gd;
            followTime = 0f;
            havePrevWall = false;
        }
    }

    // ---------------- wall following ----------------
    (float, float) FollowWall(float f, float l, float r, float ls, float rs,
                              float fc, float lc, float rc, float lsc, float rsc)
    {
        int s = followSide;
        float wallSide = s > 0 ? ls : rs;
        float wallDiag = s > 0 ? l : r;
        float otherSide = s > 0 ? rs : ls;
        float otherDiagC = s > 0 ? rc : lc;
        float wallDiagC = s > 0 ? lc : rc;
        float otherSideC = s > 0 ? rsc : lsc;

        float dWall = Mathf.Min(wallSide, wallDiag * 0.72f);

        float target = WALL_TARGET;
        if (otherSide < 0.5f) target = Mathf.Min(WALL_TARGET, 0.5f * (wallSide + otherSide));

        float turn, speed;

        if (fc < BLOCK_FRONT || wallDiagC < 0.16f)
        {
            // inner corner (or wall corner poking into the path): turn away
            turn = -s * 1.0f;
            speed = (fc > 0.32f && wallDiagC > 0.16f && otherDiagC > 0.2f) ? 0.12f : 0f;
            havePrevWall = false;
        }
        else if (dWall > WALL_LOST)
        {
            // outer corner / wall lost: arc around towards the wall side
            turn = s * 0.7f;
            speed = Mathf.Min(0.4f, SafeSpeed(fc, lc, rc));
            havePrevWall = false;
        }
        else
        {
            float err = dWall - target;
            float deriv = 0f;
            if (havePrevWall) deriv = Mathf.Clamp((dWall - prevWall) / DT, -0.5f, 0.5f);
            prevWall = dWall;
            havePrevWall = true;

            turn = s * (3.0f * err + 1.2f * deriv);

            // keep away from the opposite side as well
            if (otherDiagC < 0.28f) turn += s * (0.28f - otherDiagC) * 3.5f;
            if (otherSideC < 0.12f) turn += s * (0.12f - otherSideC) * 4f;
            // wall diagonal getting close: bias away from wall
            if (wallDiagC < 0.25f) turn -= s * (0.25f - wallDiagC) * 4f;

            turn = Mathf.Clamp(turn, -0.9f, 0.9f);
            speed = Mathf.Min(0.85f, SafeSpeed(fc, lc, rc));
            speed *= 1f - 0.5f * Mathf.Abs(turn);
        }

        return (Mathf.Clamp01(speed), Mathf.Clamp(turn, -1f, 1f));
    }

    // ---------------- DistBug leave condition ----------------
    bool CanLeave(float f, float l, float r, float ls, float rs, float gd, float ga)
    {
        if (followTime < 1.0f) return false;
        float a = Mathf.Abs(ga);
        if (a > 100f) return false;

        float free;
        if (a <= 22.5f) free = Mathf.Min(f, Mathf.Min(l, r) + 0.3f);
        else if (a <= 67.5f) free = ga > 0f ? Mathf.Min(l, f + 0.2f) : Mathf.Min(r, f + 0.2f);
        else free = ga > 0f ? Mathf.Min(ls, l + 0.2f) : Mathf.Min(rs, r + 0.2f);

        bool goalTowardWall = (followSide > 0 && ga > 30f) || (followSide < 0 && ga < -30f);
        if (goalTowardWall && free < 0.7f) return false;

        free *= 0.9f; // noise margin

        if (free < BLOCK_FRONT + 0.2f && free < gd) return false;

        if (free >= gd) return true;
        if (gd - free <= hitDist - LEAVE_STEP) return true;
        return false;
    }
}
