using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "Adaptive Rover Navigation";

    // Navigation states
    private enum NavState
    {
        GoalSeeking,
        AvoidLeft,
        AvoidRight
    }

    private NavState state;

    // Filtered sensor readings
    private float fFront, fLeft, fRight;
    private float fLeftSide, fRightSide;
    private bool initialized;

    // Navigation memory
    private float previousGoalDistance;
    private float bestGoalDistance;
    private float noProgressTime;
    private float stateTime;
    private float lastTurn;
    private int escapeCount;

    // Timing: controller updates at 20 Hz
    private const float DT = 0.05f;

    // Rover safety parameters
    private const float SAFE_DISTANCE = 0.29f;
    private const float CRITICAL_DISTANCE = 0.22f;
    private const float WALL_TARGET = 0.37f;

    private const float MAX_SPEED = 1.0f;
    private const float MIN_SPEED = 0.13f;

    private const float DEG2RAD = Mathf.Deg2Rad;

    public void ResetController()
    {
        state = NavState.GoalSeeking;

        fFront = 4f;
        fLeft = 4f;
        fRight = 4f;
        fLeftSide = 4f;
        fRightSide = 4f;

        initialized = false;

        previousGoalDistance = 0f;
        bestGoalDistance = 999f;

        noProgressTime = 0f;
        stateTime = 0f;
        lastTurn = 0f;

        escapeCount = 0;
    }

    // Keep sensor values within valid limits.
    private float Clean(float value)
    {
        if (float.IsNaN(value) || float.IsInfinity(value))
            return 4f;

        return Mathf.Clamp(value, 0.02f, 4f);
    }

    // Exponential moving average with protection against
    // occasional false maximum-range readings.
    private float Filter(float oldValue, float newValue)
    {
        newValue = Clean(newValue);

        if (newValue >= 3.95f && oldValue < 2.5f)
            newValue = oldValue;

        return Mathf.Lerp(oldValue, newValue, 0.35f);
    }

    private void UpdateSensors(Observation obs)
    {
        if (!initialized)
        {
            fFront = Clean(obs.front);
            fLeft = Clean(obs.left);
            fRight = Clean(obs.right);
            fLeftSide = Clean(obs.leftSide);
            fRightSide = Clean(obs.rightSide);

            initialized = true;
            return;
        }

        fFront = Filter(fFront, obs.front);
        fLeft = Filter(fLeft, obs.left);
        fRight = Filter(fRight, obs.right);

        fLeftSide = Filter(fLeftSide, obs.leftSide);
        fRightSide = Filter(fRightSide, obs.rightSide);
    }

    private float ClampTurn(float value)
    {
        return Mathf.Clamp(value, -1f, 1f);
    }

    private float NormalizeAngle(float angle)
    {
        while (angle > 180f) angle -= 360f;
        while (angle < -180f) angle += 360f;

        return angle;
    }

    // Convert goal bearing into a normalized steering command.
    private float GoalSteering(float angle)
    {
        angle = NormalizeAngle(angle);

        return Mathf.Clamp(angle / 65f, -1f, 1f);
    }

    // Estimate obstacle danger from a sensor.
    private float Danger(float distance)
    {
        if (distance >= SAFE_DISTANCE)
            return 0f;

        return Mathf.Clamp01(
            (SAFE_DISTANCE - distance) /
            (SAFE_DISTANCE - 0.12f)
        );
    }

    // Decide which side has more available clearance.
    private bool PreferLeft(float goalAngle)
    {
        float leftClearance =
            0.55f * fLeft +
            0.45f * fLeftSide;

        float rightClearance =
            0.55f * fRight +
            0.45f * fRightSide;

        // Slightly favor the direction of the goal.
        leftClearance += Mathf.Clamp(goalAngle / 180f, 0f, 0.3f);
        rightClearance += Mathf.Clamp(-goalAngle / 180f, 0f, 0.3f);

        return leftClearance >= rightClearance;
    }

    // Reactive obstacle avoidance while seeking the goal.
    private float AvoidanceSteering(float goalAngle)
    {
        float steering = GoalSteering(goalAngle);

        float frontDanger = Danger(fFront);
        float leftDanger = Danger(fLeft);
        float rightDanger = Danger(fRight);

        float leftSideDanger = Danger(fLeftSide);
        float rightSideDanger = Danger(fRightSide);

        // Repulsion from forward obstacles.
        steering += 0.95f * leftDanger;
        steering -= 0.95f * rightDanger;

        // Side obstacle repulsion.
        steering -= 0.55f * leftSideDanger;
        steering += 0.55f * rightSideDanger;

        // A close frontal obstacle requires a stronger turn.
        if (frontDanger > 0.15f)
        {
            float direction = PreferLeft(goalAngle) ? 1f : -1f;

            steering += direction *
                        (0.65f + 0.45f * frontDanger);
        }

        // Increase steering as the front gets closer.
        steering *= 1f + 0.35f * frontDanger;

        return ClampTurn(steering);
    }

    // Follow a wall to navigate around obstacles and U-shaped traps.
    private float WallFollowingSteering()
    {
        float steering;

        if (state == NavState.AvoidLeft)
        {
            // Left-hand wall following.
            // Too close: steer right.
            // Too far: steer left.
            float error = WALL_TARGET - fLeftSide;

            steering = -1.5f * error;

            // Front-left obstacle: steer right.
            steering -= 0.9f * Danger(fLeft);

            // Front obstacle: steer right more strongly.
            steering -= 1.1f * Danger(fFront);

            // Right-side clearance provides additional correction.
            if (fRightSide < 0.25f)
                steering -= 0.25f;
        }
        else
        {
            // Right-hand wall following.
            // Too close: steer left.
            // Too far: steer right.
            float error = WALL_TARGET - fRightSide;

            steering = 1.5f * error;

            // Front-right obstacle: steer left.
            steering += 0.9f * Danger(fRight);

            // Front obstacle: steer left more strongly.
            steering += 1.1f * Danger(fFront);

            if (fLeftSide < 0.25f)
                steering += 0.25f;
        }

        return ClampTurn(steering);
    }

    // Detect when it is safe to leave wall following.
    private bool CanReturnToGoal(float goalAngle)
    {
        bool frontClear = fFront > 0.65f;

        bool directionClear;

        if (goalAngle >= 0f)
            directionClear = fLeft > 0.55f;
        else
            directionClear = fRight > 0.55f;

        bool goalAligned = Mathf.Abs(goalAngle) < 35f;

        bool enoughTime = stateTime > 1.0f;

        return frontClear &&
               directionClear &&
               goalAligned &&
               enoughTime;
    }

    // Update progress monitoring.
    private void UpdateProgress(float goalDistance)
    {
        if (goalDistance < bestGoalDistance - 0.035f)
        {
            bestGoalDistance = goalDistance;
            noProgressTime = 0f;
        }
        else
        {
            noProgressTime += DT;
        }

        if (goalDistance < previousGoalDistance - 0.02f)
            noProgressTime = Mathf.Max(0f, noProgressTime - DT);

        previousGoalDistance = goalDistance;
    }

    // Escape behavior when navigation is not making progress.
    private void HandleStuck(float goalAngle)
    {
        if (noProgressTime < 3.0f)
            return;

        escapeCount++;

        // Change wall-following direction when stuck.
        if (state == NavState.AvoidLeft)
            state = NavState.AvoidRight;
        else if (state == NavState.AvoidRight)
            state = NavState.AvoidLeft;
        else
            state = PreferLeft(goalAngle)
                ? NavState.AvoidLeft
                : NavState.AvoidRight;

        stateTime = 0f;
        noProgressTime = 0f;
    }

    // Main controller.
    public (float speed01, float turn) Decide(Observation obs)
    {
        UpdateSensors(obs);

        float goalDistance = Mathf.Max(0f, obs.goalDistance);
        float goalAngle = NormalizeAngle(obs.goalAngleDeg);

        stateTime += DT;

        UpdateProgress(goalDistance);

        // Goal reached or sufficiently close.
        if (goalDistance < 0.18f)
        {
            return (0f, 0f);
        }

        // Detect a nearby frontal obstacle.
        bool obstacleAhead =
            fFront < 0.48f ||
            fLeft < 0.32f ||
            fRight < 0.32f;

        // Enter obstacle avoidance mode.
        if (state == NavState.GoalSeeking && obstacleAhead)
        {
            state = PreferLeft(goalAngle)
                ? NavState.AvoidLeft
                : NavState.AvoidRight;

            stateTime = 0f;
        }

        // Check whether the rover can safely return to goal seeking.
        if (state != NavState.GoalSeeking &&
            CanReturnToGoal(goalAngle))
        {
            state = NavState.GoalSeeking;
            stateTime = 0f;
            noProgressTime = 0f;
        }

        // Stuck detection and recovery.
        HandleStuck(goalAngle);

        float turn;

        if (state == NavState.GoalSeeking)
        {
            turn = AvoidanceSteering(goalAngle);
        }
        else
        {
            turn = WallFollowingSteering();
        }

        // Emergency steering for extremely close obstacles.
        if (fFront < CRITICAL_DISTANCE)
        {
            if (fLeft > fRight)
                turn = 1f;
            else
                turn = -1f;
        }

        // Smooth steering to reduce oscillations.
        turn = Mathf.Lerp(lastTurn, turn, 0.65f);
        lastTurn = turn;

        turn = ClampTurn(turn);

        // Adaptive speed based on obstacle clearance.
        float minimumDistance = Mathf.Min(
            fFront,
            fLeft,
            fRight,
            fLeftSide,
            fRightSide
        );

        float speed = MAX_SPEED;

        if (minimumDistance < 0.8f)
        {
            speed = Mathf.Lerp(
                0.18f,
                0.65f,
                Mathf.Clamp01((minimumDistance - 0.20f) / 0.60f)
            );
        }

        // Reduce speed when making sharp turns.
        speed *= 1f - 0.55f * Mathf.Abs(turn);

        // Reduce speed when the goal is directly behind.
        if (Mathf.Abs(goalAngle) > 110f)
            speed *= 0.65f;

        // Slow down further in wall-following mode.
        if (state != NavState.GoalSeeking)
            speed = Mathf.Min(speed, 0.55f);

        // Avoid moving forward into a very close obstacle.
        if (fFront < 0.17f)
            speed = 0f;

        speed = Mathf.Clamp(speed, 0f, 1f);

        return (speed, turn);
    }

The previous controller has three weaknesses:

Steering oscillation: It continuously changes steering based on noisy left/right sensor readings.

Wall-following instability: It tries to maintain a fixed wall distance, but box corners and U-shaped obstacles cause repeated direction changes.

Poor escape strategy: It switches between left and right wall-following modes without remembering which direction it has already explored.

I recommend changing the controller to a persistent-direction obstacle avoidance strategy.

The important changes are:

Commit to one avoidance direction instead of reconsidering it every frame.

Add steering hysteresis to prevent rapid left/right switching.

Use stronger corner detection.

Follow obstacles until there is a clear route toward the goal.

Detect lack of progress and change strategy only after a sustained period.

Reduce speed when turning sharply or navigating close to obstacles.

Here is the revised controller. It uses a persistent wall-following direction and a timed recovery state rather than continuously changing its mind.

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "Persistent Bug Navigation V2";

    private enum Mode
    {
        Seek,
        FollowLeft,
        FollowRight,
        Escape
    }

    private Mode mode;

    private const float DT = 0.05f;
    private const float MAX_RANGE = 4f;

    private const float FOLLOW_DISTANCE = 0.38f;
    private const float FRONT_LIMIT = 0.52f;
    private const float CRITICAL_LIMIT = 0.23f;

    private float front, left, right;
    private float leftSide, rightSide;

    private bool initialized;

    // Persistent navigation direction
    private int followDirection;
    private int escapeDirection;

    // Timers
    private float modeTime;
    private float clearTime;
    private float stuckTime;
    private float escapeTime;

    // Progress monitoring
    private float bestDistance;
    private float previousDistance;

    // Steering memory
    private float previousTurn;

    // Sensor filtering
    private int frontClearCount;
    private int leftClearCount;
    private int rightClearCount;
    private int leftSideClearCount;
    private int rightSideClearCount;

    public void ResetController()
    {
        mode = Mode.Seek;

        front = MAX_RANGE;
        left = MAX_RANGE;
        right = MAX_RANGE;
        leftSide = MAX_RANGE;
        rightSide = MAX_RANGE;

        initialized = false;

        followDirection = 1;
        escapeDirection = 1;

        modeTime = 0f;
        clearTime = 0f;
        stuckTime = 0f;
        escapeTime = 0f;

        bestDistance = 999f;
        previousDistance = 999f;

        previousTurn = 0f;

        frontClearCount = 0;
        leftClearCount = 0;
        rightClearCount = 0;
        leftSideClearCount = 0;
        rightSideClearCount = 0;
    }

    private float Clean(float v)
    {
        if (float.IsNaN(v) || float.IsInfinity(v))
            return MAX_RANGE;

        return Mathf.Clamp(v, 0.02f, MAX_RANGE);
    }

    // Reject isolated false maximum-range readings.
    private float Filter(
        float oldValue,
        float newValue,
        ref int clearCount)
    {
        newValue = Clean(newValue);

        if (newValue >= 3.95f)
        {
            clearCount++;

            if (clearCount < 3 && oldValue < 3.0f)
                return oldValue;
        }
        else
        {
            clearCount = 0;
        }

        return Mathf.Lerp(oldValue, newValue, 0.40f);
    }

    private void UpdateSensors(Observation o)
    {
        if (!initialized)
        {
            front = Clean(o.front);
            left = Clean(o.left);
            right = Clean(o.right);

            leftSide = Clean(o.leftSide);
            rightSide = Clean(o.rightSide);

            initialized = true;
            return;
        }

        front = Filter(front, o.front, ref frontClearCount);
        left = Filter(left, o.left, ref leftClearCount);
        right = Filter(right, o.right, ref rightClearCount);

        leftSide = Filter(
            leftSide, o.leftSide, ref leftSideClearCount);

        rightSide = Filter(
            rightSide, o.rightSide, ref rightSideClearCount);
    }

    private float ClampTurn(float v)
    {
        return Mathf.Clamp(v, -1f, 1f);
    }

    private float NormalizeAngle(float a)
    {
        while (a > 180f) a -= 360f;
        while (a < -180f) a += 360f;

        return a;
    }

    private float Danger(float distance)
    {
        return Mathf.Clamp01((0.62f - distance) / 0.48f);
    }

    // Select direction only when entering obstacle avoidance.
    private int SelectDirection()
    {
        float leftClearance =
            0.65f * left +
            0.35f * leftSide;

        float rightClearance =
            0.65f * right +
            0.35f * rightSide;

        if (leftClearance > rightClearance + 0.12f)
            return 1;

        if (rightClearance > leftClearance + 0.12f)
            return -1;

        // Break ties consistently.
        return leftClearance >= rightClearance ? 1 : -1;
    }

    private void EnterFollow(int direction)
    {
        followDirection = direction;

        mode = direction > 0
            ? Mode.FollowLeft
            : Mode.FollowRight;

        modeTime = 0f;
        clearTime = 0f;
        stuckTime = 0f;
    }

    private float SeekSteering(float goalAngle)
    {
        float turn = Mathf.Clamp(goalAngle / 65f, -1f, 1f);

        // Forward obstacle repulsion.
        turn += 0.85f * Danger(left);
        turn -= 0.85f * Danger(right);

        // Side obstacle repulsion.
        turn -= 0.45f * Danger(leftSide);
        turn += 0.45f * Danger(rightSide);

        return ClampTurn(turn);
    }

    // Persistent wall following.
    private float FollowSteering()
    {
        float turn;

        if (followDirection > 0)
        {
            // Follow the obstacle on the left.
            float error = leftSide - FOLLOW_DISTANCE;

            turn = -1.45f * error;

            // Front-left obstacle.
            turn -= 0.85f * Danger(left);

            // Directly ahead.
            turn -= 1.15f * Danger(front);

            // Obstacle on the opposite side.
            turn += 0.35f * Danger(rightSide);

            // Lost the left wall: gently search for it.
            if (leftSide > 1.1f)
                turn += 0.25f;
        }
        else
        {
            // Follow the obstacle on the right.
            float error = rightSide - FOLLOW_DISTANCE;

            turn = 1.45f * error;

            turn += 0.85f * Danger(right);
            turn += 1.15f * Danger(front);

            turn -= 0.35f * Danger(leftSide);

            if (rightSide > 1.1f)
                turn -= 0.25f;
        }

        return ClampTurn(turn);
    }

    // Check whether the path toward the goal is clear.
    private bool GoalPathClear(float angle)
    {
        if (front < 0.72f)
            return false;

        if (Mathf.Abs(angle) < 30f)
            return front > 0.90f;

        if (angle > 0f)
            return left > 0.85f;

        return right > 0.85f;
    }

    private void UpdateProgress(float distance)
    {
        if (distance < bestDistance - 0.06f)
        {
            bestDistance = distance;
            stuckTime = 0f;
        }
        else
        {
            stuckTime += DT;
        }

        if (distance < previousDistance - 0.025f)
            stuckTime = Mathf.Max(0f, stuckTime - 0.5f * DT);

        previousDistance = distance;
    }

    private void StartEscape()
    {
        // Change direction once after sustained lack of progress.
        escapeDirection = -followDirection;

        if (mode == Mode.Seek)
            escapeDirection = -SelectDirection();

        mode = Mode.Escape;

        escapeTime = 0f;
        modeTime = 0f;
        clearTime = 0f;
        stuckTime = 0f;
    }

    private float EscapeSteering()
    {
        // Turn toward the selected open side.
        float turn = escapeDirection;

        // Avoid turning into a close obstacle.
        if (escapeDirection > 0 && left < 0.32f)
            turn = -0.5f;

        if (escapeDirection < 0 && right < 0.32f)
            turn = 0.5f;

        return ClampTurn(turn);
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        UpdateSensors(obs);

        float goalDistance = Mathf.Max(0f, obs.goalDistance);
        float goalAngle = NormalizeAngle(obs.goalAngleDeg);

        modeTime += DT;

        UpdateProgress(goalDistance);

        // Goal reached.
        if (goalDistance < 0.16f)
            return (0f, 0f);

        bool obstacleAhead =
            front < FRONT_LIMIT ||
            left < 0.34f ||
            right < 0.34f;

        // Enter obstacle avoidance.
        if (mode == Mode.Seek && obstacleAhead)
        {
            EnterFollow(SelectDirection());
        }

        // Escape mode: hold the selected direction.
        if (mode == Mode.Escape)
        {
            escapeTime += DT;

            if (escapeTime > 1.1f)
            {
                EnterFollow(escapeDirection);
            }
        }

        // Check for a clear route back to the goal.
        if (mode == Mode.FollowLeft ||
            mode == Mode.FollowRight)
        {
            if (GoalPathClear(goalAngle))
                clearTime += DT;
            else
                clearTime = 0f;

            // Require sustained clearance to prevent oscillation.
            if (clearTime > 0.65f && modeTime > 1.5f)
            {
                mode = Mode.Seek;

                modeTime = 0f;
                clearTime = 0f;
                stuckTime = 0f;

                bestDistance = goalDistance;
            }
        }

        // Persistent lack-of-progress recovery.
        if (mode != Mode.Escape && stuckTime > 5.0f)
        {
            StartEscape();
        }

        float turn;

        if (mode == Mode.Seek)
        {
            turn = SeekSteering(goalAngle);
        }
        else if (mode == Mode.Escape)
        {
            turn = EscapeSteering();
        }
        else
        {
            turn = FollowSteering();
        }

        // Emergency collision avoidance.
        if (front < CRITICAL_LIMIT)
        {
            if (left > right)
                turn = 1f;
            else
                turn = -1f;
        }

        // Prevent sudden steering reversals.
        float smoothing = mode == Mode.Escape ? 0.85f : 0.40f;

        turn = Mathf.Lerp(previousTurn, turn, smoothing);

        previousTurn = ClampTurn(turn);

        turn = previousTurn;

        // Adaptive speed.
        float nearest = Mathf.Min(
            front,
            left,
            right,
            leftSide,
            rightSide
        );

        float speed = 0.85f;

        if (nearest < 0.85f)
        {
            speed = Mathf.Lerp(
                0.12f,
                0.65f,
                Mathf.Clamp01((nearest - 0.18f) / 0.67f)
            );
        }

        // Reduce speed for sharp turns.
        speed *= 1f - 0.65f * Mathf.Abs(turn);

        // Wall following requires slower movement.
        if (mode == Mode.FollowLeft ||
            mode == Mode.FollowRight)
        {
            speed = Mathf.Min(speed, 0.48f);
        }

        // Escape at a controlled speed.
        if (mode == Mode.Escape)
            speed = 0.20f;

        // Critical proximity.
        if (front < 0.15f)
            speed = 0f;

        return (
            Mathf.Clamp(speed, 0f, 1f),
            ClampTurn(turn)
        );
    }

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "Rover Navigation V3";

    private enum State
    {
        Seek,
        TurnAroundObstacle,
        FollowWall,
        EscapeTurn
    }

    private State state;

    private const float DT = 0.05f;

    private const float SAFE_FRONT = 0.58f;
    private const float SAFE_DIAGONAL = 0.40f;
    private const float CRITICAL = 0.25f;
    private const float WALL_TARGET = 0.42f;

    private float front, left, right;
    private float leftSide, rightSide;

    private bool initialized;

    private int direction;
    private int turnDirection;

    private float stateTimer;
    private float clearTimer;
    private float stuckTimer;
    private float escapeTimer;

    private float bestDistance;
    private float previousDistance;

    private float previousTurn;
    private float filteredGoalAngle;

    private int frontClear;
    private int leftClear;
    private int rightClear;
    private int leftSideClear;
    private int rightSideClear;

    private int recoveryAttempts;

    public void ResetController()
    {
        state = State.Seek;

        front = left = right = 4f;
        leftSide = rightSide = 4f;

        initialized = false;

        direction = 1;
        turnDirection = 1;

        stateTimer = 0f;
        clearTimer = 0f;
        stuckTimer = 0f;
        escapeTimer = 0f;

        bestDistance = 999f;
        previousDistance = 999f;

        previousTurn = 0f;
        filteredGoalAngle = 0f;

        frontClear = 0;
        leftClear = 0;
        rightClear = 0;
        leftSideClear = 0;
        rightSideClear = 0;

        recoveryAttempts = 0;
    }

    private float Clean(float v)
    {
        if (float.IsNaN(v) || float.IsInfinity(v))
            return 4f;

        return Mathf.Clamp(v, 0.02f, 4f);
    }

    private float Filter(
        float oldValue,
        float newValue,
        ref int clearCount)
    {
        newValue = Clean(newValue);

        // Reject isolated false maximum readings.
        if (newValue >= 3.95f && oldValue < 2.5f)
        {
            clearCount++;

            if (clearCount < 3)
                return oldValue;
        }
        else
        {
            clearCount = 0;
        }

        return Mathf.Lerp(oldValue, newValue, 0.35f);
    }

    private void UpdateSensors(Observation o)
    {
        if (!initialized)
        {
            front = Clean(o.front);
            left = Clean(o.left);
            right = Clean(o.right);
            leftSide = Clean(o.leftSide);
            rightSide = Clean(o.rightSide);

            filteredGoalAngle = o.goalAngleDeg;

            initialized = true;
            return;
        }

        front = Filter(front, o.front, ref frontClear);
        left = Filter(left, o.left, ref leftClear);
        right = Filter(right, o.right, ref rightClear);

        leftSide = Filter(
            leftSide, o.leftSide, ref leftSideClear);

        rightSide = Filter(
            rightSide, o.rightSide, ref rightSideClear);

        filteredGoalAngle = Mathf.LerpAngle(
            filteredGoalAngle,
            o.goalAngleDeg,
            0.35f
        );
    }

    private float Danger(float d)
    {
        return Mathf.Clamp01((0.65f - d) / 0.48f);
    }

    private float ClampTurn(float t)
    {
        return Mathf.Clamp(t, -1f, 1f);
    }

    // Choose a side only when starting a new obstacle encounter.
    private int ChooseDirection()
    {
        float leftSpace =
            0.55f * left + 0.45f * leftSide;

        float rightSpace =
            0.55f * right + 0.45f * rightSide;

        if (leftSpace > rightSpace + 0.10f)
            return 1;

        if (rightSpace > leftSpace + 0.10f)
            return -1;

        return leftSpace >= rightSpace ? 1 : -1;
    }

    private void BeginAvoidance(int dir)
    {
        direction = dir;

        // Turn away from the obstacle while establishing
        // a consistent wall-following side.
        turnDirection = -direction;

        state = State.TurnAroundObstacle;

        stateTimer = 0f;
        clearTimer = 0f;
    }

    private bool ObstacleAhead()
    {
        return front < SAFE_FRONT ||
               left < SAFE_DIAGONAL ||
               right < SAFE_DIAGONAL;
    }

    // Goal-seeking steering.
    private float SeekSteering(float angle)
    {
        float turn = Mathf.Clamp(angle / 65f, -1f, 1f);

        turn += 0.80f * Danger(left);
        turn -= 0.80f * Danger(right);

        turn -= 0.45f * Danger(leftSide);
        turn += 0.45f * Danger(rightSide);

        return ClampTurn(turn);
    }

    // Corrected wall-following controller.
    private float FollowSteering()
    {
        float turn;

        if (direction > 0)
        {
            // Keep obstacle on left.
            // Too far from wall -> turn left.
            // Too close -> turn right.

            turn = 1.7f * (leftSide - WALL_TARGET);

            // Front-left obstacle.
            turn -= 0.65f * Danger(left);

            // Obstacle directly ahead.
            turn -= 1.15f * Danger(front);

            // Maintain clearance on opposite side.
            turn += 0.30f * Danger(rightSide);
        }
        else
        {
            // Keep obstacle on right.
            // Too far from wall -> turn right.
            // Too close -> turn left.

            turn = -1.7f * (rightSide - WALL_TARGET);

            turn += 0.65f * Danger(right);

            turn += 1.15f * Danger(front);

            turn -= 0.30f * Danger(leftSide);
        }

        return ClampTurn(turn);
    }

    // A goal-facing direction must remain clear before
    // leaving obstacle-following mode.
    private bool GoalDirectionClear(float angle)
    {
        if (front < 0.85f)
            return false;

        if (Mathf.Abs(angle) < 25f)
            return front > 1.0f;

        if (angle > 0f)
            return left > 0.85f;

        return right > 0.85f;
    }

    private void UpdateProgress(float distance)
    {
        if (distance < bestDistance - 0.08f)
        {
            bestDistance = distance;
            stuckTimer = 0f;
            recoveryAttempts = 0;
        }
        else
        {
            stuckTimer += DT;
        }

        previousDistance = distance;
    }

    private void StartRecovery()
    {
        // Only switch direction after sustained lack of progress.
        direction = -direction;

        turnDirection = -direction;

        state = State.EscapeTurn;

        escapeTimer = 0f;
        stateTimer = 0f;
        stuckTimer = 0f;
        clearTimer = 0f;

        recoveryAttempts++;
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        UpdateSensors(obs);

        float distance = Mathf.Max(0f, obs.goalDistance);
        float angle = Mathf.Clamp(filteredGoalAngle, -180f, 180f);

        stateTimer += DT;

        UpdateProgress(distance);

        // Goal reached.
        if (distance < 0.17f)
            return (0f, 0f);

        // State 1: Seek the goal.
        if (state == State.Seek)
        {
            if (ObstacleAhead())
                BeginAvoidance(ChooseDirection());
        }

        // State 2: Initial controlled turn.
        if (state == State.TurnAroundObstacle)
        {
            if (stateTimer >= 0.85f)
            {
                state = State.FollowWall;
                stateTimer = 0f;
                clearTimer = 0f;
            }
        }

        // State 3: Persistent wall following.
        if (state == State.FollowWall)
        {
            if (GoalDirectionClear(angle))
                clearTimer += DT;
            else
                clearTimer = 0f;

            // Require sustained clearance.
            if (clearTimer >= 0.8f &&
                stateTimer >= 1.5f)
            {
                state = State.Seek;

                stateTimer = 0f;
                clearTimer = 0f;
                stuckTimer = 0f;

                bestDistance = distance;
            }
        }

        // State 4: Recovery turn.
        if (state == State.EscapeTurn)
        {
            escapeTimer += DT;

            if (escapeTimer >= 0.75f)
            {
                state = State.FollowWall;

                stateTimer = 0f;
                clearTimer = 0f;
            }
        }

        // Recovery only after sustained lack of progress.
        if (state != State.EscapeTurn &&
            state != State.TurnAroundObstacle &&
            stuckTimer >= 6f &&
            recoveryAttempts < 3)
        {
            StartRecovery();
        }

        float turn = 0f;
        float speed = 0.75f;

        switch (state)
        {
            case State.Seek:

                turn = SeekSteering(angle);
                speed = 0.75f;

                break;

            case State.TurnAroundObstacle:

                turn = turnDirection * 0.75f;
                speed = 0f;

                break;

            case State.FollowWall:

                turn = FollowSteering();
                speed = 0.38f;

                break;

            case State.EscapeTurn:

                turn = turnDirection * 0.85f;
                speed = 0f;

                break;
        }

        // Emergency collision avoidance.
        if (front < CRITICAL)
        {
            turn = left > right ? 1f : -1f;
            speed = 0.08f;
        }

        // Smooth ordinary steering, but do not smooth
        // fixed-direction turning states.
        if (state == State.Seek ||
            state == State.FollowWall)
        {
            turn = Mathf.Lerp(previousTurn, turn, 0.55f);
        }

        previousTurn = ClampTurn(turn);

        // Slow down near obstacles.
        float nearest = Mathf.Min(
            front, left, right, leftSide, rightSide
        );

        if (nearest < 0.8f)
        {
            float clearanceSpeed = Mathf.Lerp(
                0.12f,
                0.55f,
                Mathf.Clamp01((nearest - 0.20f) / 0.60f)
            );

            speed = Mathf.Min(speed, clearanceSpeed);
        }

        // Slow down on sharp turns.
        if (speed > 0f)
            speed *= 1f - 0.55f * Mathf.Abs(turn);

        // Never drive into an extremely close frontal obstacle.
        if (front < 0.16f)
            speed = 0f;

        return (
            Mathf.Clamp(speed, 0f, 1f),
            ClampTurn(turn)
        );
    }
}

using System;
using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "Rover V5 Anti-Oscillation";

    private enum Mode
    {
        Seek,
        Avoid,
        Escape
    }

    private Mode mode;

    private const float DT = 0.05f;
    private const float SAFE_FRONT = 0.55f;
    private const float SAFE_DIAGONAL = 0.34f;
    private const float CRITICAL = 0.20f;

    private float front, left, right;
    private float leftSide, rightSide;

    private float previousTurn;
    private float filteredAngle;

    private bool initialized;

    // Fixed avoidance direction:
    // +1 = left, -1 = right.
    private int avoidDirection;
    private int escapeDirection;

    // Steering reversal protection.
    private int steeringSign;
    private int pendingSign;
    private float reversalTimer;

    // State timers.
    private float stateTimer;
    private float clearTimer;
    private float stuckTimer;
    private float escapeTimer;

    // Progress monitoring.
    private float progressReference;
    private float progressTimer;

    private int recoveryCount;

    // Sensor false-reading protection.
    private int fc, lc, rc, lsc, rsc;

    public void ResetController()
    {
        mode = Mode.Seek;

        front = left = right = 4f;
        leftSide = rightSide = 4f;

        previousTurn = 0f;
        filteredAngle = 0f;

        initialized = false;

        avoidDirection = 1;
        escapeDirection = 1;

        steeringSign = 0;
        pendingSign = 0;
        reversalTimer = 0f;

        stateTimer = 0f;
        clearTimer = 0f;
        stuckTimer = 0f;
        escapeTimer = 0f;

        progressReference = 999f;
        progressTimer = 0f;

        recoveryCount = 0;

        fc = lc = rc = lsc = rsc = 0;
    }

    private float Clean(float x)
    {
        if (float.IsNaN(x) || float.IsInfinity(x))
            return 4f;

        return Mathf.Clamp(x, 0.02f, 4f);
    }

    // Asymmetric filtering:
    // React quickly to nearby obstacles.
    // React slowly to increasing distance.
    // Reject isolated false 4m readings.
    private float Filter(
        float oldValue,
        float newValue,
        ref int clearCount)
    {
        newValue = Clean(newValue);

        if (newValue >= 3.95f && oldValue < 2.5f)
        {
            clearCount++;

            if (clearCount < 3)
                return oldValue;
        }
        else
        {
            clearCount = 0;
        }

        float alpha = newValue < oldValue ? 0.65f : 0.22f;

        return Mathf.Lerp(oldValue, newValue, alpha);
    }

    private void UpdateSensors(Observation o)
    {
        if (!initialized)
        {
            front = Clean(o.front);
            left = Clean(o.left);
            right = Clean(o.right);

            leftSide = Clean(o.leftSide);
            rightSide = Clean(o.rightSide);

            filteredAngle = o.goalAngleDeg;

            initialized = true;
            return;
        }

        front = Filter(front, o.front, ref fc);
        left = Filter(left, o.left, ref lc);
        right = Filter(right, o.right, ref rc);

        leftSide = Filter(leftSide, o.leftSide, ref lsc);
        rightSide = Filter(rightSide, o.rightSide, ref rsc);

        filteredAngle = Mathf.LerpAngle(
            filteredAngle,
            o.goalAngleDeg,
            0.25f
        );
    }

    private float Danger(float d)
    {
        return Mathf.Clamp01((0.65f - d) / 0.45f);
    }

    private float ClampTurn(float t)
    {
        return Mathf.Clamp(t, -1f, 1f);
    }

    private float GoalTurn(float angle)
    {
        // Deadband prevents left-right switching
        // when the goal is almost straight ahead.
        if (Mathf.Abs(angle) < 8f)
            return 0f;

        return Mathf.Clamp(angle / 65f, -0.9f, 0.9f);
    }

    private int ChooseDirection()
    {
        float l = 0.6f * left + 0.4f * leftSide;
        float r = 0.6f * right + 0.4f * rightSide;

        if (l > r + 0.12f)
            return 1;

        if (r > l + 0.12f)
            return -1;

        return l >= r ? 1 : -1;
    }

    private void EnterAvoid(int direction)
    {
        avoidDirection = direction;

        mode = Mode.Avoid;

        stateTimer = 0f;
        clearTimer = 0f;
        stuckTimer = 0f;

        progressReference = 999f;

        // Reset steering reversal memory on state change.
        steeringSign = 0;
        pendingSign = 0;
        reversalTimer = 0f;
    }

    // Seek toward the goal while avoiding nearby obstacles.
    private float SeekSteering(float angle)
    {
        float turn = GoalTurn(angle);

        turn += 0.65f * Danger(left);
        turn -= 0.65f * Danger(right);

        turn -= 0.35f * Danger(leftSide);
        turn += 0.35f * Danger(rightSide);

        return ClampTurn(turn);
    }

    // Persistent direction commitment while passing obstacles.
    private float AvoidSteering(float angle)
    {
        float goal = GoalTurn(angle);

        // Maintain the selected passing direction.
        float turn = avoidDirection * 0.48f;

        // Gradually consider the goal direction.
        turn += goal * 0.25f;

        // Keep clearance from forward obstacles.
        turn += 0.70f * Danger(left);
        turn -= 0.70f * Danger(right);

        // Side clearance correction.
        if (avoidDirection > 0)
        {
            if (leftSide < 0.65f)
                turn -= 0.35f * Danger(leftSide);

            if (front < 0.75f)
                turn -= 0.35f;
        }
        else
        {
            if (rightSide < 0.65f)
                turn += 0.35f * Danger(rightSide);

            if (front < 0.75f)
                turn += 0.35f;
        }

        return ClampTurn(turn);
    }

    // Prevent rapid steering direction reversals.
    private float ApplyHysteresis(float desired)
    {
        if (Mathf.Abs(desired) < 0.08f)
        {
            pendingSign = 0;
            reversalTimer = 0f;

            return Mathf.Lerp(previousTurn, 0f, 0.25f);
        }

        int desiredSign = desired > 0f ? 1 : -1;

        if (steeringSign == 0)
        {
            steeringSign = desiredSign;
        }

        if (desiredSign != steeringSign)
        {
            // Require a sustained opposite steering request.
            if (pendingSign != desiredSign)
            {
                pendingSign = desiredSign;
                reversalTimer = 0f;
            }
            else
            {
                reversalTimer += DT;
            }

            // Emergency-level steering can reverse immediately.
            if (Mathf.Abs(desired) > 0.88f)
            {
                steeringSign = desiredSign;
                pendingSign = 0;
                reversalTimer = 0f;
            }
            else if (reversalTimer >= 0.30f)
            {
                steeringSign = desiredSign;
                pendingSign = 0;
                reversalTimer = 0f;
            }
            else
            {
                // Hold the existing steering direction.
                desired = steeringSign *
                    Mathf.Min(Mathf.Abs(previousTurn), 0.35f);
            }
        }
        else
        {
            pendingSign = 0;
            reversalTimer = 0f;
        }

        return Mathf.Lerp(previousTurn, desired, 0.38f);
    }

    private bool PathClear(float angle)
    {
        if (front < 0.95f)
            return false;

        if (Mathf.Abs(angle) < 30f)
            return true;

        if (angle > 0f)
            return left > 0.85f;

        return right > 0.85f;
    }

    private void UpdateProgress(float distance)
    {
        progressTimer += DT;

        if (distance < progressReference - 0.12f)
        {
            progressReference = distance;
            stuckTimer = 0f;
            recoveryCount = 0;
        }
        else
        {
            stuckTimer += DT;
        }

        if (progressTimer >= 2f)
        {
            progressTimer = 0f;

            // Update reference periodically without
            // requiring every sample to improve.
            if (distance < progressReference)
                progressReference = distance;
        }
    }

    private void StartEscape()
    {
        escapeDirection = -avoidDirection;

        if (mode == Mode.Seek)
            escapeDirection = -ChooseDirection();

        mode = Mode.Escape;

        escapeTimer = 0f;
        stuckTimer = 0f;
        stateTimer = 0f;
        clearTimer = 0f;

        recoveryCount++;

        steeringSign = escapeDirection;
        pendingSign = 0;
        reversalTimer = 0f;
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        UpdateSensors(obs);

        float distance = Mathf.Max(0f, obs.goalDistance);
        float angle = Mathf.Clamp(filteredAngle, -180f, 180f);

        stateTimer += DT;

        UpdateProgress(distance);

        if (distance < 0.16f)
            return (0f, 0f);

        // Raw readings provide immediate collision protection.
        float rawFront = Clean(obs.front);
        float rawLeft = Clean(obs.left);
        float rawRight = Clean(obs.right);

        bool obstacleAhead =
            front < SAFE_FRONT ||
            left < SAFE_DIAGONAL ||
            right < SAFE_DIAGONAL;

        // SEEK
        if (mode == Mode.Seek && obstacleAhead)
        {
            EnterAvoid(ChooseDirection());
            progressReference = distance;
        }

        // AVOID
        if (mode == Mode.Avoid)
        {
            if (PathClear(angle))
                clearTimer += DT;
            else
                clearTimer = 0f;

            if (clearTimer >= 0.75f && stateTimer >= 1.2f)
            {
                mode = Mode.Seek;

                stateTimer = 0f;
                clearTimer = 0f;
                stuckTimer = 0f;

                steeringSign = 0;
                pendingSign = 0;
                reversalTimer = 0f;

                progressReference = distance;
            }
        }

        // ESCAPE
        if (mode == Mode.Escape)
        {
            escapeTimer += DT;

            if (escapeTimer >= 0.55f)
            {
                EnterAvoid(escapeDirection);
                progressReference = distance;
            }
        }

        // Sustained stuck detection.
        if (mode != Mode.Escape &&
            stuckTimer > 5f &&
            recoveryCount < 4)
        {
            StartEscape();
        }

        float desiredTurn;
        float speed;

        if (mode == Mode.Seek)
        {
            desiredTurn = SeekSteering(angle);
            speed = 0.75f;
        }
        else if (mode == Mode.Avoid)
        {
            desiredTurn = AvoidSteering(angle);
            speed = 0.42f;
        }
        else
        {
            desiredTurn = escapeDirection * 0.80f;
            speed = 0.12f;
        }

        // Immediate safety override.
        bool emergency =
            rawFront < CRITICAL ||
            rawLeft < 0.16f ||
            rawRight < 0.16f;

        if (emergency)
        {
            if (rawFront < CRITICAL)
                desiredTurn = left > right ? 1f : -1f;
            else if (rawLeft < 0.16f)
                desiredTurn = -1f;
            else
                desiredTurn = 1f;

            speed = 0f;

            steeringSign = desiredTurn > 0f ? 1 : -1;
            pendingSign = 0;
            reversalTimer = 0f;
        }
        else
        {
            desiredTurn = ApplyHysteresis(desiredTurn);
        }

        previousTurn = ClampTurn(desiredTurn);

        // Adaptive speed.
        float nearest = Mathf.Min(
            front,
            left,
            right,
            leftSide,
            rightSide
        );

        if (nearest < 0.90f)
        {
            float clearanceSpeed = Mathf.Lerp(
                0.10f,
                0.65f,
                Mathf.Clamp01((nearest - 0.20f) / 0.70f)
            );

            speed = Mathf.Min(speed, clearanceSpeed);
        }

        // Reduce speed for sharp steering.
        speed *= 1f - 0.45f * Mathf.Abs(previousTurn);

        if (rawFront < 0.15f)
            speed = 0f;

        return (
            Mathf.Clamp(speed, 0f, 1f),
            ClampTurn(previousTurn)
        );
    }

using UnityEngine;

public class LLM_Controller : IRoverController
{
    public string Name => "Rover V6 Adaptive Navigation";

    private enum NavigationMode
    {
        GoalSeeking,
        ObstacleAvoidance,
        Recovery
    }

    private NavigationMode mode;

    private const float DT = 0.05f;
    private const float MAX_SENSOR = 4f;

    private const float FRONT_DANGER = 0.65f;
    private const float DIAGONAL_DANGER = 0.42f;
    private const float CRITICAL_DISTANCE = 0.24f;

    private const float DESIRED_WALL_DISTANCE = 0.40f;

    private float front;
    private float left;
    private float right;
    private float leftSide;
    private float rightSide;

    private bool initialized;

    private float filteredGoalAngle;

    // Navigation direction:
    // +1 = pass obstacle on its left side.
    // -1 = pass obstacle on its right side.
    private int avoidanceDirection;

    private int escapeDirection;

    // Steering memory
    private float previousTurn;
    private int steeringDirection;
    private int pendingDirection;
    private float reversalTimer;

    // Timers
    private float modeTimer;
    private float clearTimer;
    private float stuckTimer;
    private float recoveryTimer;

    // Progress tracking
    private float bestGoalDistance;
    private float progressReference;

    private int recoveryCount;

    // Sensor filtering counters
    private int fc, lc, rc, lsc, rsc;

    public void ResetController()
    {
        mode = NavigationMode.GoalSeeking;

        front = left = right = 4f;
        leftSide = rightSide = 4f;

        initialized = false;

        filteredGoalAngle = 0f;

        avoidanceDirection = 1;
        escapeDirection = 1;

        previousTurn = 0f;
        steeringDirection = 0;
        pendingDirection = 0;
        reversalTimer = 0f;

        modeTimer = 0f;
        clearTimer = 0f;
        stuckTimer = 0f;
        recoveryTimer = 0f;

        bestGoalDistance = 999f;
        progressReference = 999f;

        recoveryCount = 0;

        fc = lc = rc = lsc = rsc = 0;
    }

    private float Clean(float value)
    {
        if (float.IsNaN(value) || float.IsInfinity(value))
            return MAX_SENSOR;

        return Mathf.Clamp(value, 0.02f, MAX_SENSOR);
    }

    // Fast response to closer obstacles.
    // Slower response when obstacles disappear.
    // Reject isolated false maximum readings.
    private float Filter(
        float previous,
        float current,
        ref int clearCount)
    {
        current = Clean(current);

        if (current >= 3.95f && previous < 2.5f)
        {
            clearCount++;

            if (clearCount < 3)
                return previous;
        }
        else
        {
            clearCount = 0;
        }

        float alpha = current < previous ? 0.65f : 0.25f;

        return Mathf.Lerp(previous, current, alpha);
    }

    private void UpdateSensors(Observation obs)
    {
        if (!initialized)
        {
            front = Clean(obs.front);
            left = Clean(obs.left);
            right = Clean(obs.right);

            leftSide = Clean(obs.leftSide);
            rightSide = Clean(obs.rightSide);

            filteredGoalAngle = obs.goalAngleDeg;

            initialized = true;
            return;
        }

        front = Filter(front, obs.front, ref fc);
        left = Filter(left, obs.left, ref lc);
        right = Filter(right, obs.right, ref rc);

        leftSide = Filter(leftSide, obs.leftSide, ref lsc);
        rightSide = Filter(rightSide, obs.rightSide, ref rsc);

        filteredGoalAngle = Mathf.LerpAngle(
            filteredGoalAngle,
            obs.goalAngleDeg,
            0.35f
        );
    }

    private float ClampTurn(float value)
    {
        return Mathf.Clamp(value, -1f, 1f);
    }

    private float Danger(float distance)
    {
        return Mathf.Clamp01(
            (0.75f - distance) / 0.55f
        );
    }

    private float GoalSteering(float angle)
    {
        // Small deadband reduces unnecessary corrections.
        if (Mathf.Abs(angle) < 7f)
            return 0f;

        return Mathf.Clamp(angle / 60f, -1f, 1f);
    }

    // Choose the direction with greater available clearance.
    private int ChooseAvoidanceDirection()
    {
        float leftClearance =
            0.65f * left +
            0.35f * leftSide;

        float rightClearance =
            0.65f * right +
            0.35f * rightSide;

        if (leftClearance > rightClearance + 0.12f)
            return 1;

        if (rightClearance > leftClearance + 0.12f)
            return -1;

        return leftClearance >= rightClearance ? 1 : -1;
    }

    private void EnterAvoidance(int direction)
    {
        avoidanceDirection = direction;

        mode = NavigationMode.ObstacleAvoidance;

        modeTimer = 0f;
        clearTimer = 0f;
        stuckTimer = 0f;

        steeringDirection = 0;
        pendingDirection = 0;
        reversalTimer = 0f;
    }

    // Combined goal attraction and obstacle repulsion.
    private float CalculateGoalSteering(float angle)
    {
        float turn = GoalSteering(angle);

        turn += 0.95f * Danger(left);
        turn -= 0.95f * Danger(right);

        turn -= 0.40f * Danger(leftSide);
        turn += 0.40f * Danger(rightSide);

        return ClampTurn(turn);
    }

    // Avoidance with persistent passing direction.
    private float CalculateAvoidanceSteering(float angle)
    {
        float goalTurn = GoalSteering(angle);

        float turn = goalTurn * 0.35f;

        if (avoidanceDirection > 0)
        {
            // Prefer passing the obstacle on the left.
            // Turn away from close frontal-left obstacles.
            turn += 0.45f;
            turn -= 0.95f * Danger(left);
            turn -= 0.65f * Danger(front);

            // Maintain clearance from the left wall.
            if (leftSide < 0.85f)
            {
                turn += Mathf.Clamp(
                    (leftSide - DESIRED_WALL_DISTANCE) * 0.65f,
                    -0.35f,
                    0.35f
                );
            }
        }
        else
        {
            // Prefer passing the obstacle on the right.
            turn -= 0.45f;
            turn += 0.95f * Danger(right);
            turn += 0.65f * Danger(front);

            if (rightSide < 0.85f)
            {
                turn -= Mathf.Clamp(
                    (rightSide - DESIRED_WALL_DISTANCE) * 0.65f,
                    -0.35f,
                    0.35f
                );
            }
        }

        // Additional clearance protection.
        turn += 0.30f * Danger(leftSide);
        turn -= 0.30f * Danger(rightSide);

        return ClampTurn(turn);
    }

    // Prevent rapid left-right steering reversals.
    private float StabilizeSteering(float desired)
    {
        int sign = desired > 0.06f ? 1 :
                   desired < -0.06f ? -1 : 0;

        if (sign == 0)
        {
            pendingDirection = 0;
            reversalTimer = 0f;

            return Mathf.Lerp(previousTurn, 0f, 0.35f);
        }

        if (steeringDirection == 0)
            steeringDirection = sign;

        if (sign != steeringDirection)
        {
            if (pendingDirection != sign)
            {
                pendingDirection = sign;
                reversalTimer = 0f;
            }

            reversalTimer += DT;

            // Allow emergency steering immediately.
            if (Mathf.Abs(desired) > 0.90f)
            {
                steeringDirection = sign;
                pendingDirection = 0;
                reversalTimer = 0f;
            }
            else if (reversalTimer >= 0.25f)
            {
                steeringDirection = sign;
                pendingDirection = 0;
                reversalTimer = 0f;
            }
            else
            {
                desired = steeringDirection *
                    Mathf.Min(Mathf.Abs(previousTurn), 0.35f);
            }
        }
        else
        {
            pendingDirection = 0;
            reversalTimer = 0f;
        }

        return Mathf.Lerp(previousTurn, desired, 0.45f);
    }

    // Check whether the goal direction is sufficiently clear.
    private bool GoalPathIsClear(float angle)
    {
        if (front < 0.90f)
            return false;

        if (Mathf.Abs(angle) < 25f)
            return true;

        if (angle > 0f)
            return left > 0.75f;

        return right > 0.75f;
    }

    private void UpdateProgress(float distance)
    {
        if (distance < bestGoalDistance - 0.10f)
        {
            bestGoalDistance = distance;
            stuckTimer = 0f;
            recoveryCount = 0;
        }
        else
        {
            stuckTimer += DT;
        }

        if (distance < progressReference - 0.12f)
        {
            progressReference = distance;
        }
    }

    private void StartRecovery()
    {
        escapeDirection = -avoidanceDirection;

        if (mode == NavigationMode.GoalSeeking)
            escapeDirection = -ChooseAvoidanceDirection();

        mode = NavigationMode.Recovery;

        recoveryTimer = 0f;
        modeTimer = 0f;
        stuckTimer = 0f;
        clearTimer = 0f;

        recoveryCount++;

        steeringDirection = escapeDirection;
        pendingDirection = 0;
        reversalTimer = 0f;
    }

    public (float speed01, float turn) Decide(Observation obs)
    {
        UpdateSensors(obs);

        float goalDistance = Mathf.Max(0f, obs.goalDistance);

        float goalAngle = Mathf.Clamp(
            filteredGoalAngle,
            -180f,
            180f
        );

        modeTimer += DT;

        UpdateProgress(goalDistance);

        // Goal reached.
        if (goalDistance < 0.16f)
            return (0f, 0f);

        // Use raw sensor readings for immediate safety.
        float rawFront = Clean(obs.front);
        float rawLeft = Clean(obs.left);
        float rawRight = Clean(obs.right);

        bool obstacleAhead =
            front < FRONT_DANGER ||
            left < DIAGONAL_DANGER ||
            right < DIAGONAL_DANGER;

        // GOAL SEEKING
        if (mode == NavigationMode.GoalSeeking &&
            obstacleAhead)
        {
            EnterAvoidance(ChooseAvoidanceDirection());
        }

        // OBSTACLE AVOIDANCE
        if (mode == NavigationMode.ObstacleAvoidance)
        {
            if (GoalPathIsClear(goalAngle))
                clearTimer += DT;
            else
                clearTimer = 0f;

            if (clearTimer >= 0.65f && modeTimer >= 1.0f)
            {
                mode = NavigationMode.GoalSeeking;

                modeTimer = 0f;
                clearTimer = 0f;
                stuckTimer = 0f;

                steeringDirection = 0;
                pendingDirection = 0;
                reversalTimer = 0f;

                bestGoalDistance = goalDistance;
            }
        }

        // RECOVERY
        if (mode == NavigationMode.Recovery)
        {
            recoveryTimer += DT;

            if (recoveryTimer >= 0.50f)
            {
                EnterAvoidance(escapeDirection);
            }
        }

        // Detect lack of meaningful progress.
        if (mode != NavigationMode.Recovery &&
            stuckTimer >= 5f &&
            recoveryCount < 4)
        {
            StartRecovery();
        }

        float desiredTurn;
        float speed;

        switch (mode)
        {
            case NavigationMode.GoalSeeking:

                desiredTurn = CalculateGoalSteering(goalAngle);
                speed = 0.75f;

                break;

            case NavigationMode.ObstacleAvoidance:

                desiredTurn = CalculateAvoidanceSteering(goalAngle);
                speed = 0.45f;

                break;

            default:

                desiredTurn = escapeDirection * 0.85f;
                speed = 0.10f;

                break;
        }

        // Emergency response overrides steering hysteresis.
        bool emergency =
            rawFront < CRITICAL_DISTANCE ||
            rawLeft < 0.15f ||
            rawRight < 0.15f;

        if (emergency)
        {
            if (rawFront < CRITICAL_DISTANCE)
                desiredTurn = left > right ? 1f : -1f;
            else if (rawLeft < 0.15f)
                desiredTurn = -1f;
            else
                desiredTurn = 1f;

            speed = 0f;

            steeringDirection = desiredTurn > 0f ? 1 : -1;
            pendingDirection = 0;
            reversalTimer = 0f;
        }
        else
        {
            desiredTurn = StabilizeSteering(desiredTurn);
        }

        previousTurn = ClampTurn(desiredTurn);

        // Adaptive speed based on minimum clearance.
        float nearest = Mathf.Min(
            front,
            left,
            right,
            leftSide,
            rightSide
        );

        if (nearest < 0.90f)
        {
            float safeSpeed = Mathf.Lerp(
                0.10f,
                0.65f,
                Mathf.Clamp01((nearest - 0.20f) / 0.70f)
            );

            speed = Mathf.Min(speed, safeSpeed);
        }

        // Slow down for sharp turns.
        speed *= 1f - 0.45f * Mathf.Abs(previousTurn);

        if (rawFront < 0.16f)
            speed = 0f;

        return (
            Mathf.Clamp(speed, 0f, 1f),
            ClampTurn(previousTurn)
        );
    }
}

My recommendation for testing
Do not change the arena or controller parameters between tests. Run the same 10 scenarios.

I would pay particular attention to whether the rover now follows a smooth arc around obstacles rather than alternating left and right.

One final engineering note: the five sensors provide instantaneous distances, not obstacle geometry. A fully reliable solution for arbitrary U-shaped obstacles is impossible to guarantee with these inputs alone. V6 is a heuristic improvement, not a proven 5/10 or 10/10 solution.
